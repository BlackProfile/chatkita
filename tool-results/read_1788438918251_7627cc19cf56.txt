   153→const fmtUptime = (ms: number): string => {
   154→  const total = Math.max(0, Math.floor(ms / 1000));
   155→  const d = Math.floor(total / 86400);
   156→  const h = Math.floor((total % 86400) / 3600);
   157→  const m = Math.floor((total % 3600) / 60);
   158→  if (d > 0) return `${d} hari ${h} jam`;
   159→  if (h > 0) return `${h} jam ${m} menit`;
   160→  return `${m} menit`;
   161→};
   162→
   163→const fmtDay = (iso: string): string => {
   164→  const [, m, d] = iso.split("-");
   165→  return `${d}/${m}`;
   166→};
   167→
   168→/** v13 — dd/mm dari ISO datetime penuh (aman utk joinedAt dll). */
   169→const fmtIsoDay = (iso: string): string => {
   170→  const d = new Date(iso);
   171→  if (Number.isNaN(d.getTime())) return "—";
   172→  return d.toLocaleDateString("id-ID", { day: "2-digit", month: "2-digit" });
   173→};
   174→
   175→const fmtDateTime = (iso: string): string => {
   176→  const d = new Date(iso);
   177→  if (Number.isNaN(d.getTime())) return "—";
   178→  return d.toLocaleString("id-ID", { dateStyle: "medium", timeStyle: "short" });
   179→};
   180→
   181→/** v13 — durasi respons human-readable (ms → detik/menit/jam). */
   182→const fmtLag = (ms: number): string => {
   183→  if (ms < 60_000) return `${Math.round(ms / 1000)} detik`;
   184→  if (ms < 3_600_000) return `${Math.round(ms / 60_000)} menit`;
   185→  return `${(ms / 3_600_000).toFixed(1)} jam`;
   186→};

   255→  className,
   256→}: {

   271→              style={{ height: `${Math.max(3, (v / max) * 100)}%` }}
   272→              title={`${labels[i]}: ${v} pesan`}

   274→          </div>
