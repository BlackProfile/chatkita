   377→/** Fire-and-forget push to every stored subscription of `userId`. */
   378→const pushSend = async (userId: string, payload: PushPayload) => {
   379→  if (!VAPID_PUBLIC) return
   380→  try {
   381→    const subs = db
   382→      .query('SELECT * FROM push_subscriptions WHERE user_id = ?')
   383→      .all(userId) as Array<{ endpoint: string; p256dh: string; auth: string }>
   384→    if (subs.length === 0) return
   385→    await Promise.all(
   386→      subs.map(async (s) => {
   387→        try {
   388→          await webpush.sendNotification(
   389→            { endpoint: s.endpoint, keys: { p256dh: s.p256dh, auth: s.auth } },
   390→            JSON.stringify(payload),
   391→            { TTL: 3600 }
   392→          )
   393→        } catch (err) {
   394→          const status = (err as { statusCode?: number }).statusCode
   395→          if (status === 404 || status === 410) {
   396→            db.run('DELETE FROM push_subscriptions WHERE endpoint = ?', [s.endpoint])
   397→          }
   398→        }
   399→      })
   400→    )
   401→  } catch (err) {
   402→    console.error('Push send error:', (err as Error)?.message ?? err)
   403→  }
   404→}

   460→/** Grouped reactions for one message (message:updated payloads). */
   461→const reactionsFor = (messageId: number): { emoji: string; userIds: string[] }[] => {
   462→  const rows = db
   463→    .query('SELECT user_id, emoji FROM message_reactions WHERE message_id = ?')
   464→    .all(messageId) as Array<{ user_id: string; emoji: string }>
   465→  const byEmoji = new Map<string, string[]>()
   466→  for (const r of rows) {
   467→    const users = byEmoji.get(r.emoji) ?? []
   468→    users.push(r.user_id)
   469→    byEmoji.set(r.emoji, users)
   470→  }
   471→  return [...byEmoji.entries()].map(([emoji, userIds]) => ({ emoji, userIds }))
   472→}

   474→/** Attach grouped reactions to a batch of messages (single query). */
   475→const attachReactions = (rows: MessageRow[], messages: ChatMessageApi[]) => {
   476→  if (rows.length === 0) return
   477→  const placeholders = rows.map(() => '?').join(',')
   478→  const reRows = db
   479→    .query(
   480→      `SELECT message_id, user_id, emoji FROM message_reactions WHERE message_id IN (${placeholders})`
   481→    )
   482→    .all(...rows.map((r) => r.id)) as Array<{ message_id: number; user_id: string; emoji: string }>
   483→  if (reRows.length === 0) return
   484→  const byMsg = new Map<number, Map<string, string[]>>()
   485→  for (const r of reRows) {
   486→    let byEmoji = byMsg.get(r.message_id)
   487→    if (!byEmoji) {
   488→      byEmoji = new Map()
   489→      byMsg.set(r.message_id, byEmoji)
   490→    }
   491→    const users = byEmoji.get(r.emoji) ?? []
   492→    users.push(r.user_id)
   493→    byEmoji.set(r.emoji, users)
   494→  }
   495→  for (const msg of messages) {
   496→    const byEmoji = byMsg.get(msg.id)
   497→    if (byEmoji && byEmoji.size > 0) {
   498→      msg.reactions = [...byEmoji.entries()].map(([emoji, userIds]) => ({ emoji, userIds }))
   499→    }
   500→  }
   501→}
