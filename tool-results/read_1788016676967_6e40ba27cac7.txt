    47→    47→interface StoredUser {
    48→    48→  userId: string;
    49→    49→  name: string;
    50→    50→}
    51→    51→
    52→    52→/** Read (and validate) the persisted login. Client-side only. */
    53→    53→function readStoredUser(): StoredUser | null {
    54→    54→  try {
    55→    55→    const raw = window.localStorage.getItem(MESSENGER_STORAGE_KEY);
    56→    56→    if (!raw) return null;
    57→    57→    const parsed = JSON.parse(raw) as Partial<StoredUser> | null;
    58→    58→    if (
    59→    59→      parsed &&
    60→    60→      typeof parsed.userId === "string" &&
    61→    61→      typeof parsed.name === "string" &&
    62→    62→      parsed.userId.length > 0 &&
    63→    63→      parsed.name.length > 0
    64→    64→    ) {
    65→    65→      return { userId: parsed.userId, name: parsed.name };
    66→    66→    }
    67→    67→    return null;
    68→    68→  } catch {
    69→    69→    return null;
    70→    70→  }
    71→    71→}
    72→    72→
    73→    73→/**
    74→    74→ * ChatKita Messenger — Telegram-style 1-on-1 chat:
    75→    75→ * login by name, conversation list, user search, real-time messaging,
    76→    76→ * typing indicators, unread counts, and online presence.
    77→    77→ * Owns its own socket connection, always disconnected on unmount.
    78→    78→ */
    79→    79→export function Messenger() {
    80→    80→  const isMobile = useIsMobile();
    81→    81→
    82→    82→  const [me, setMe] = useState<StoredUser | null>(null);
    83→    83→  const [conversations, setConversations] = useState<ConversationOverview[]>([]);
    84→    84→  const [activeId, setActiveId] = useState<string | null>(null);
    85→    85→  const [messagesMap, setMessagesMap] = useState<Record<string, ChatMessage[]>>({});
    86→    86→  const [typingMap, setTypingMap] = useState<Record<string, boolean>>({});
    87→    87→  const [connected, setConnected] = useState(false);
    88→    88→  const [socket, setSocket] = useState<Socket | null>(null);
    89→    89→  const [name, setName] = useState("");
    90→    90→  const [authError, setAuthError] = useState<string | null>(null);
    91→    91→  const [filter, setFilter] = useState("");
    92→    92→  const [newChatOpen, setNewChatOpen] = useState(false);
    93→    93→  const [input, setInput] = useState("");
    94→    94→  const [sendError, setSendError] = useState(false);
    95→    95→  /** Bumped on logout to tear down + recreate the socket (fresh rooms). */
    96→    96→  const [epoch, setEpoch] = useState(0);
    97→    97→
    98→    98→  const socketRef = useRef<Socket | null>(null);
    99→    99→  const meRef = useRef<StoredUser | null>(null);
   100→   100→  const activeIdRef = useRef<string | null>(null);
   101→   101→  const typingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   102→   102→  const partnerTypingTimersRef = useRef<Record<string, ReturnType<typeof setTimeout>>>({});
   103→   103→  const scrollRef = useRef<HTMLDivElement | null>(null);
   104→   104→
   105→   105→  /* ---------------------------------------------------------------- */
   106→   106→  /* Socket lifecycle (recreated on logout via `epoch`)                */
   107→   107→  /* ---------------------------------------------------------------- */
   108→   108→  useEffect(() => {
   109→   109→    // Restore persisted login (ref only) so the first `connect` can
   110→   110→    // re-auth immediately. State is set inside the connect ack.
   111→   111→    meRef.current = readStoredUser();
   112→   112→
   113→   113→    const socket = createChatSocket();
   114→   114→    socketRef.current = socket;
   115→   115→
   116→   116→    socket.on("connect", () => {
   117→   117→      setConnected(true);
   118→   118→      setSocket(socket);
   119→   119→      const current = meRef.current;
   120→   120→      if (!current) return;
   121→   121→      // Re-auth on EVERY connect to (re)join the personal room.
   122→   122→      socket.emit(
   123→   123→        "user:auth",
   124→   124→        { name: current.name, userId: current.userId },
   125→   125→        (res: UserAuthAck | ChatErrorAck) => {
   126→   126→          if (res.ok) {
   127→   127→            const next = { userId: res.user.id, name: res.user.name };
   128→   128→            meRef.current = next;
   129→   129→            setMe(next);
   130→   130→            setConversations(res.conversations);
   131→   131→          } else {
   132→   132→            // Stored user no longer valid — drop it, back to login form.
   133→   133→            window.localStorage.removeItem(MESSENGER_STORAGE_KEY);
   134→   134→            meRef.current = null;
   135→   135→            setMe(null);
   136→   136→            setConversations([]);
   137→   137→            activeIdRef.current = null;
   138→   138→            setActiveId(null);
   139→   139→            setAuthError("Sesi berakhir. Silakan masuk kembali.");
   140→   140→          }
   141→   141→        }
   142→   142→      );
   143→   143→    });
   144→   144→
   145→   145→    socket.on("disconnect", () => {
   146→   146→      setConnected(false);
   147→   147→      setSocket(null);
   148→   148→    });
   149→   149→
   150→   150→    socket.on("conversations:update", (list: ConversationOverview[]) => {
   151→   151→      setConversations(list);
   152→   152→    });
   153→   153→
   154→   154→    // Append messages ONLY here; skip if the last stored message already
   155→   155→    // has the same id (history-replacement vs broadcast race).
   156→   156→    socket.on("message:new", (msg: ChatMessage) => {
   157→   157→      setMessagesMap((prev) => {
   158→   158→        const list = prev[msg.conversationId];
   159→   159→        if (!list || list.length === 0) {
   160→   160→          return { ...prev, [msg.conversationId]: [msg] };
   161→   161→        }
   162→   162→        const last = list[list.length - 1];
   163→   163→        if (last.id === msg.id) return prev;
   164→   164→        return { ...prev, [msg.conversationId]: [...list, msg] };
   165→   165→      });
   166→   166→      if (msg.conversationId === activeIdRef.current) {
   167→   167→        socketRef.current?.emit("messages:read", { conversationId: msg.conversationId });
   168→   168→      }
   169→   169→    });
   170→   170→
   171→   171→    socket.on(
   172→   172→      "partner:typing",
   173→   173→      ({ conversationId, isTyping }: { conversationId: string; isTyping: boolean }) => {
   174→   174→        const timers = partnerTypingTimersRef.current;
   175→   175→        if (timers[conversationId]) {
   176→   176→          clearTimeout(timers[conversationId]);
   177→   177→          delete timers[conversationId];
   178→   178→        }
   179→   179→        setTypingMap((prev) => {
   180→   180→          const next = { ...prev };
   181→   181→          if (isTyping) next[conversationId] = true;
   182→   182→          else delete next[conversationId];
   183→   183→          return next;
   184→   184→        });
   185→   185→        if (isTyping) {
   186→   186→          // Auto-clear after 4s in case a stop event is missed.
   187→   187→          timers[conversationId] = setTimeout(() => {
   188→   188→            delete partnerTypingTimersRef.current[conversationId];
   189→   189→            setTypingMap((prev) => {
   190→   190→              if (!prev[conversationId]) return prev;
   191→   191→              const next = { ...prev };
   192→   192→              delete next[conversationId];
   193→   193→              return next;
   194→   194→            });
   195→   195→          }, 4000);
   196→   196→        }
   197→   197→      }
   198→   198→    );
   199→   199→
   200→   200→    socket.on("presence:update", ({ userId, online }: { userId: string; online: boolean }) => {
   201→   201→      setConversations((prev) =>
   202→   202→        prev.map((c) =>
   203→   203→          c.partner.id === userId ? { ...c, partner: { ...c.partner, online } } : c
   204→   204→        )
   205→   205→      );
   206→   206→    });
   207→   207→
   208→   208→    return () => {
   209→   209→      if (typingTimerRef.current) clearTimeout(typingTimerRef.current);
   210→   210→      Object.values(partnerTypingTimersRef.current).forEach((t) => clearTimeout(t));
   211→   211→      partnerTypingTimersRef.current = {};
   212→   212→      socket.disconnect();
   213→   213→      socketRef.current = null;
   214→   214→    };
   215→   215→  }, [epoch]);
   216→   216→
   217→   217→  /* ---------------------------------------------------------------- */
   218→   218→  /* Derived values                                                    */
   219→   219→  /* ---------------------------------------------------------------- */
   220→   220→  const filteredConversations = useMemo(() => {
   221→   221→    const q = filter.trim().toLowerCase();
   222→   222→    if (!q) return conversations;
   223→   223→    return conversations.filter((c) => c.partner.name.toLowerCase().includes(q));
   224→   224→  }, [conversations, filter]);
   225→   225→
   226→   226→  const activeConversation: ConversationOverview | null = activeId
   227→   227→    ? conversations.find((c) => c.id === activeId) ?? null
   228→   228→    : null;
   229→   229→  const activeMessages = activeId ? messagesMap[activeId] ?? [] : [];
   230→   230→  const activeTyping = activeId ? typingMap[activeId] === true : false;
   231→   231→  const totalUnread = conversations.reduce((sum, c) => sum + c.unread, 0);
   232→   232→
   233→   233→  /* ---------------------------------------------------------------- */
   234→   234→  /* Auto-scroll to latest message (scroll the viewport, never the page) */
   235→   235→  /* ---------------------------------------------------------------- */
   236→   236→  useEffect(() => {
   237→   237→    const root = scrollRef.current;
   238→   238→    if (!root) return;
   239→   239→    const viewport = root.querySelector<HTMLDivElement>(
   240→   240→      "[data-radix-scroll-area-viewport]"
   241→   241→    );
   242→   242→    if (viewport) viewport.scrollTop = viewport.scrollHeight;
   243→   243→  }, [activeId, activeMessages, activeTyping]);
   244→   244→
   245→   245→  /* ---------------------------------------------------------------- */
   246→   246→  /* Actions                                                           */
   247→   247→  /* ---------------------------------------------------------------- */
   248→   248→  const handleAuth = () => {
   249→   249→    const socket = socketRef.current;
   250→   250→    const trimmed = name.trim();
   251→   251→    if (!socket || !connected || !trimmed) return;
   252→   252→    setAuthError(null);
   253→   253→    socket.emit("user:auth", { name: trimmed }, (res: UserAuthAck | ChatErrorAck) => {
   254→   254→      if (res.ok) {
   255→   255→        const next = { userId: res.user.id, name: res.user.name };
   256→   256→        window.localStorage.setItem(MESSENGER_STORAGE_KEY, JSON.stringify(next));
   257→   257→        meRef.current = next;
   258→   258→        setMe(next);
   259→   259→        setConversations(res.conversations);
   260→   260→      } else {
   261→   261→        setAuthError(
   262→   262→          res.error === "INVALID_NAME"
   263→   263→            ? "Nama tidak valid (1–40 karakter)."
   264→   264→            : "Terjadi kesalahan, coba lagi."
   265→   265→        );
   266→   266→      }
   267→   267→    });
   268→   268→  };
   269→   269→
   270→   270→  const handleLogout = () => {
   271→   271→    window.localStorage.removeItem(MESSENGER_STORAGE_KEY);
   272→   272→    meRef.current = null;
   273→   273→    activeIdRef.current = null;
   274→   274→    setMe(null);
   275→   275→    setConversations([]);
   276→   276→    setMessagesMap({});
   277→   277→    setTypingMap({});
   278→   278→    setActiveId(null);
   279→   279→    setFilter("");
   280→   280→    setAuthError(null);
   281→   281→    setInput("");
   282→   282→    setSendError(false);
   283→   283→    setNewChatOpen(false);
   284→   284→    // Fresh socket ⇒ server cleanly forgets this client's rooms.
   285→   285→    setEpoch((e) => e + 1);
   286→   286→  };
   287→   287→
   288→   288→  const handleSelectConversation = (id: string) => {
   289→   289→    const socket = socketRef.current;
   290→   290→    if (!socket) return;
   291→   291→    activeIdRef.current = id;
   292→   292→    setActiveId(id);
   293→   293→    setSendError(false);
   294→   294→    socket.emit("messages:history", { conversationId: id }, (res: HistoryAck | ChatErrorAck) => {
   295→   295→      if (res.ok) {
   296→   296→        setMessagesMap((prev) => ({ ...prev, [id]: res.messages }));
   297→   297→        socketRef.current?.emit("messages:read", { conversationId: id });
   298→   298→      }
   299→   299→    });
   300→   300→  };
   301→   301→
   302→   302→  const handleBackToList = () => {
   303→   303→    activeIdRef.current = null;
   304→   304→    setActiveId(null);
   305→   305→    setSendError(false);
   306→   306→  };
   307→   307→
   308→   308→  const handleSend = () => {
   309→   309→    const socket = socketRef.current;
   310→   310→    const id = activeIdRef.current;
   311→   311→    const content = input.trim();
   312→   312→    if (!socket || !id || !connected || !content) return;
   313→   313→    setInput("");
   314→   314→    setSendError(false);
   315→   315→    socket.emit(
   316→   316→      "messages:send",
   317→   317→      { conversationId: id, content },
   318→   318→      (res: MessageAck | ChatErrorAck) => {
   319→   319→        if (!res.ok) {
   320→   320→          setInput(content); // restore the failed text
   321→   321→          setSendError(true);
   322→   322→        }
   323→   323→      }
   324→   324→    );
   325→   325→  };
   326→   326→
   327→   327→  const handleInputChange = (value: string) => {
   328→   328→    setInput(value);
   329→   329→    setSendError(false);
   330→   330→    const socket = socketRef.current;
   331→   331→    const id = activeIdRef.current;
   332→   332→    if (!socket || !id || !connected) return;
   333→   333→    socket.emit("typing", { conversationId: id, isTyping: true });
   334→   334→    if (typingTimerRef.current) clearTimeout(typingTimerRef.current);
   335→   335→    typingTimerRef.current = setTimeout(() => {
   336→   336→      socketRef.current?.emit("typing", { conversationId: id, isTyping: false });
   337→   337→    }, 1500);
   338→   338→  };
   339→   339→
   340→   340→  /* ---------------------------------------------------------------- */
   341→   341→  /* Render: login                                                     */
   342→   342→  /* ---------------------------------------------------------------- */
   343→   343→  if (!me) {
   344→   344→    return (
   345→   345→      <div className="flex min-h-0 w-full flex-1 items-center justify-center px-4 pb-6">
   346→   346→        <Card className="w-full max-w-md rounded-2xl">
   347→   347→          <CardHeader>
   348→   348→            <span
   349→   349→              className="w-fit rounded-lg bg-emerald-600/10 p-3 text-emerald-600"
   350→   350→              aria-hidden="true"
   351→   351→            >
   352→   352→              <MessageCircleMore className="size-6" />
   353→   353→            </span>
   354→   354→            <CardTitle className="text-xl">Masuk Chat</CardTitle>
   355→   355→            <CardDescription>
   356→   356→              Masukkan nama Anda untuk melihat percakapan dan chat dengan teman
   357→   357→            </CardDescription>
   358→   358→          </CardHeader>
   359→   359→          <CardContent>
   360→   360→            <form
   361→   361→              className="space-y-3"
   362→   362→              onSubmit={(e) => {
   363→   363→                e.preventDefault();
   364→   364→                handleAuth();
   365→   365→              }}
   366→   366→            >
   367→   367→              <div className="space-y-2">
   368→   368→                <Label htmlFor="messenger-name">Nama Anda</Label>
   369→   369→                <Input
   370→   370→                  id="messenger-name"
   371→   371→                  value={name}
   372→   372→                  maxLength={MAX_NAME_LENGTH}
   373→   373→                  placeholder="cth. Budi Santoso"
   374→   374→                  autoComplete="name"
   375→   375→                  className="h-11"
   376→   376→                  onChange={(e) => {
   377→   377→                    setName(e.target.value);
   378→   378→                    setAuthError(null);
   379→   379→                  }}
   380→   380→                />
   381→   381→              </div>
   382→   382→              {authError ? <p className="text-sm text-destructive">{authError}</p> : null}
   383→   383→              <Button
   384→   384→                type="submit"
   385→   385→                className="h-11 w-full bg-emerald-600 text-white hover:bg-emerald-600/90"
   386→   386→                disabled={!connected || !name.trim()}
   387→   387→              >
   388→   388→                {connected ? "Masuk" : "Menghubungkan…"}
   389→   389→              </Button>
   390→   390→              <p className="text-center text-xs text-muted-foreground">
   391→   391→                Nama yang sama = akun yang sama, jadi Anda bisa lanjut chat kapan saja
   392→   392→              </p>
   393→   393→            </form>
   394→   394→          </CardContent>
   395→   395→        </Card>
   396→   396→      </div>
   397→   397→    );
   398→   398→  }
   399→   399→
   400→   400→  /* ---------------------------------------------------------------- */
   401→   401→  /* Render: messenger (full screen)                                   */
   402→   402→  /* ---------------------------------------------------------------- */
   403→   403→  const showSidebar = !isMobile || !activeId;
   404→   404→  const showChatPane = !isMobile || activeId !== null;
   405→   405→
   406→   406→  const partnerStatus = activeTyping
   407→   407→    ? "sedang mengetik…"
   408→   408→    : activeConversation?.partner.online
   409→   409→      ? "Online"
   410→   410→      : "Offline";
   411→   411→
   412→   412→  return (
   413→   413→    <div className="flex min-h-0 w-full flex-1 flex-col">
   414→   414→      <div className="flex min-h-0 flex-1 flex-col overflow-hidden bg-card">
   415→   415→        {/* Reconnecting strip */}
   416→   416→        {!connected ? (
   417→   417→          <p className="bg-amber-500/10 px-3 py-1.5 text-xs text-amber-600 dark:text-amber-400">
   418→   418→            Koneksi terputus — mencoba menyambung ulang…
   419→   419→          </p>
   420→   420→        ) : null}
   421→   421→
   422→   422→        <div className="grid min-h-0 flex-1 grid-cols-[minmax(0,1fr)] md:grid-cols-[340px_1fr] lg:grid-cols-[380px_1fr]">
   423→   423→          {/* ------------------------- Sidebar ------------------------- */}
   424→   424→          {showSidebar ? (
   425→   425→            <aside className="flex min-h-0 min-w-0 flex-col overflow-hidden md:border-r">
   426→   426→              {/* Profile */}
   427→   427→              <div className="flex items-center gap-3 border-b p-3">
   428→   428→                <Avatar className="size-10">
   429→   429→                  <AvatarFallback
   430→   430→                    className={cn(
   431→   431→                      "text-sm font-semibold text-white",
   432→   432→                      avatarColorClass(me.name)
   433→   433→                    )}
   434→   434→                  >
   435→   435→                    {initials(me.name)}
   436→   436→                  </AvatarFallback>
   437→   437→                </Avatar>
   438→   438→                <div className="min-w-0 flex-1">
   439→   439→                  <p className="truncate font-semibold leading-tight">{me.name}</p>
   440→   440→                  <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   441→   441→                    <span
   442→   442→                      aria-hidden="true"
   443→   443→                      className={cn(
   444→   444→                        "inline-block size-1.5 shrink-0 rounded-full",
   445→   445→                        connected ? "bg-emerald-500" : "bg-muted-foreground/40"
   446→   446→                      )}
   447→   447→                    />
   448→   448→                    {connected ? "Online" : "Menghubungkan…"}
   449→   449→                  </p>
   450→   450→                </div>
   451→   451→                <Button
   452→   452→                  variant="ghost"
   453→   453→                  size="icon"
   454→   454→                  className="size-11 shrink-0 text-muted-foreground hover:text-destructive"
   455→   455→                  aria-label="Keluar"
   456→   456→                  onClick={handleLogout}
   457→   457→                >
   458→   458→                  <LogOut className="size-4" aria-hidden="true" />
   459→   459→                </Button>
   460→   460→              </div>
   461→   461→
   462→   462→              {/* Actions */}
   463→   463→              <div className="flex items-center gap-2 p-3">
   464→   464→                <Button
   465→   465→                  className="h-11 flex-1 bg-emerald-600 text-white hover:bg-emerald-600/90"
   466→   466→                  onClick={() => setNewChatOpen(true)}
   467→   467→                >
   468→   468→                  <UserRound className="size-4" aria-hidden="true" />
   469→   469→                  Chat Baru
