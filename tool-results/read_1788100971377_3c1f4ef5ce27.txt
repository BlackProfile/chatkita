   415→   415→const getConversationsFor = (userId: string): ConversationOverviewApi[] => {
   416→   416→  const rows = db
   417→   417→    .query(
   418→   418→      `SELECT
   419→   419→        c.id,
   420→   420→        c.last_message_at,
   421→   421→        CASE WHEN c.user_a_id = $me THEN c.user_b_id ELSE c.user_a_id END AS partner_id,
   422→   422→        p.name AS partner_name,
   423→   423→        p.last_seen_at AS partner_last_seen,
   424→   424→        p.label AS partner_label,
   425→   425→        lm.id AS last_id,
   426→   426→        lm.sender_id AS last_sender,
   427→   427→        lm.content AS last_content,
   428→   428→        lm.created_at AS last_at,
   429→   429→        lm.type AS last_type,
   430→   430→        lm.deleted_at AS last_deleted,
   431→   431→        (SELECT r.last_read_message_id FROM reads r
   432→   432→          WHERE r.conversation_id = c.id
   433→   433→            AND r.user_id = (CASE WHEN c.user_a_id = $me THEN c.user_b_id ELSE c.user_a_id END)
   434→   434→        ) AS partner_read,
   435→   435→        (
   436→   436→          SELECT COUNT(*) FROM messages m
   437→   437→          WHERE m.conversation_id = c.id
   438→   438→            AND m.sender_id != $me

   535→   535→  const m = /^(\d{1,2}):(\d{2})$/.exec(s)
   536→   536→  if (!m) return fallback

   540→   540→const isWithinHours = () => {
   541→   541→  const h = settings.hours
   542→   542→  if (!h.enabled) return true
   543→   543→  const { day, minutes } = bangkokNow()
   544→   544→  if (Array.isArray(h.days) && h.days.length > 0 && !h.days.includes(day)) return false
   545→   545→  const start = parseHm(h.start, 9 * 60)
   546→   546→  const end = parseHm(h.end, 17 * 60)
   547→   547→  return start <= end ? minutes >= start && minutes < end : minutes >= start || minutes < end
   548→   548→}
