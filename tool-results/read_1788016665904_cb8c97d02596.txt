    47→interface StoredUser {
    48→  userId: string;
    49→  name: string;
    50→}
    51→
    52→/** Read (and validate) the persisted login. Client-side only. */
    53→function readStoredUser(): StoredUser | null {
    54→  try {
    55→    const raw = window.localStorage.getItem(MESSENGER_STORAGE_KEY);
    56→    if (!raw) return null;
    57→    const parsed = JSON.parse(raw) as Partial<StoredUser> | null;
    58→    if (
    59→      parsed &&
    60→      typeof parsed.userId === "string" &&
    61→      typeof parsed.name === "string" &&
    62→      parsed.userId.length > 0 &&
    63→      parsed.name.length > 0
    64→    ) {
    65→      return { userId: parsed.userId, name: parsed.name };
    66→    }
    67→    return null;
    68→  } catch {
    69→    return null;
    70→  }
    71→}
    72→
    73→/**
    74→ * ChatKita Messenger — Telegram-style 1-on-1 chat:
    75→ * login by name, conversation list, user search, real-time messaging,
    76→ * typing indicators, unread counts, and online presence.
    77→ * Owns its own socket connection, always disconnected on unmount.
    78→ */
    79→export function Messenger() {
    80→  const isMobile = useIsMobile();
    81→
    82→  const [me, setMe] = useState<StoredUser | null>(null);
    83→  const [conversations, setConversations] = useState<ConversationOverview[]>([]);
    84→  const [activeId, setActiveId] = useState<string | null>(null);
    85→  const [messagesMap, setMessagesMap] = useState<Record<string, ChatMessage[]>>({});
    86→  const [typingMap, setTypingMap] = useState<Record<string, boolean>>({});
    87→  const [connected, setConnected] = useState(false);
    88→  const [socket, setSocket] = useState<Socket | null>(null);
    89→  const [name, setName] = useState("");
    90→  const [authError, setAuthError] = useState<string | null>(null);
    91→  const [filter, setFilter] = useState("");
    92→  const [newChatOpen, setNewChatOpen] = useState(false);
    93→  const [input, setInput] = useState("");
    94→  const [sendError, setSendError] = useState(false);
    95→  /** Bumped on logout to tear down + recreate the socket (fresh rooms). */
    96→  const [epoch, setEpoch] = useState(0);
    97→
    98→  const socketRef = useRef<Socket | null>(null);
    99→  const meRef = useRef<StoredUser | null>(null);
   100→  const activeIdRef = useRef<string | null>(null);
   101→  const typingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   102→  const partnerTypingTimersRef = useRef<Record<string, ReturnType<typeof setTimeout>>>({});
   103→  const scrollRef = useRef<HTMLDivElement | null>(null);
   104→
   105→  /* ---------------------------------------------------------------- */
   106→  /* Socket lifecycle (recreated on logout via `epoch`)                */
   107→  /* ---------------------------------------------------------------- */
   108→  useEffect(() => {
   109→    // Restore persisted login (ref only) so the first `connect` can
   110→    // re-auth immediately. State is set inside the connect ack.
   111→    meRef.current = readStoredUser();
   112→
   113→    const socket = createChatSocket();
   114→    socketRef.current = socket;
   115→
   116→    socket.on("connect", () => {
   117→      setConnected(true);
   118→      setSocket(socket);
   119→      const current = meRef.current;
   120→      if (!current) return;
   121→      // Re-auth on EVERY connect to (re)join the personal room.
   122→      socket.emit(
   123→        "user:auth",
   124→        { name: current.name, userId: current.userId },
   125→        (res: UserAuthAck | ChatErrorAck) => {
   126→          if (res.ok) {
   127→            const next = { userId: res.user.id, name: res.user.name };
   128→            meRef.current = next;
   129→            setMe(next);
   130→            setConversations(res.conversations);
   131→          } else {
   132→            // Stored user no longer valid — drop it, back to login form.
   133→            window.localStorage.removeItem(MESSENGER_STORAGE_KEY);
   134→            meRef.current = null;
   135→            setMe(null);
   136→            setConversations([]);
   137→            activeIdRef.current = null;
   138→            setActiveId(null);
   139→            setAuthError("Sesi berakhir. Silakan masuk kembali.");
   140→          }
   141→        }
   142→      );
   143→    });
   144→
   145→    socket.on("disconnect", () => {
   146→      setConnected(false);
   147→      setSocket(null);
   148→    });
   149→
   150→    socket.on("conversations:update", (list: ConversationOverview[]) => {
   151→      setConversations(list);
   152→    });
   153→
   154→    // Append messages ONLY here; skip if the last stored message already
   155→    // has the same id (history-replacement vs broadcast race).
   156→    socket.on("message:new", (msg: ChatMessage) => {
   157→      setMessagesMap((prev) => {
   158→        const list = prev[msg.conversationId];
   159→        if (!list || list.length === 0) {
   160→          return { ...prev, [msg.conversationId]: [msg] };
   161→        }
   162→        const last = list[list.length - 1];
   163→        if (last.id === msg.id) return prev;
   164→        return { ...prev, [msg.conversationId]: [...list, msg] };
   165→      });
   166→      if (msg.conversationId === activeIdRef.current) {
   167→        socketRef.current?.emit("messages:read", { conversationId: msg.conversationId });
   168→      }
   169→    });
   170→
   171→    socket.on(
   172→      "partner:typing",
   173→      ({ conversationId, isTyping }: { conversationId: string; isTyping: boolean }) => {
   174→        const timers = partnerTypingTimersRef.current;
   175→        if (timers[conversationId]) {
   176→          clearTimeout(timers[conversationId]);
   177→          delete timers[conversationId];
   178→        }
   179→        setTypingMap((prev) => {
   180→          const next = { ...prev };
   181→          if (isTyping) next[conversationId] = true;
   182→          else delete next[conversationId];
   183→          return next;
   184→        });
   185→        if (isTyping) {
   186→          // Auto-clear after 4s in case a stop event is missed.
   187→          timers[conversationId] = setTimeout(() => {
   188→            delete partnerTypingTimersRef.current[conversationId];
   189→            setTypingMap((prev) => {
   190→              if (!prev[conversationId]) return prev;
   191→              const next = { ...prev };
   192→              delete next[conversationId];
   193→              return next;
   194→            });
   195→          }, 4000);
   196→        }
   197→      }
   198→    );
   199→
   200→    socket.on("presence:update", ({ userId, online }: { userId: string; online: boolean }) => {
   201→      setConversations((prev) =>
   202→        prev.map((c) =>
   203→          c.partner.id === userId ? { ...c, partner: { ...c.partner, online } } : c
   204→        )
   205→      );
   206→    });
   207→
   208→    return () => {
   209→      if (typingTimerRef.current) clearTimeout(typingTimerRef.current);
   210→      Object.values(partnerTypingTimersRef.current).forEach((t) => clearTimeout(t));
   211→      partnerTypingTimersRef.current = {};
   212→      socket.disconnect();
   213→      socketRef.current = null;
   214→    };
   215→  }, [epoch]);
   216→
   217→  /* ---------------------------------------------------------------- */
   218→  /* Derived values                                                    */
   219→  /* ---------------------------------------------------------------- */
   220→  const filteredConversations = useMemo(() => {
   221→    const q = filter.trim().toLowerCase();
   222→    if (!q) return conversations;
   223→    return conversations.filter((c) => c.partner.name.toLowerCase().includes(q));
   224→  }, [conversations, filter]);
   225→
   226→  const activeConversation: ConversationOverview | null = activeId
   227→    ? conversations.find((c) => c.id === activeId) ?? null
   228→    : null;
   229→  const activeMessages = activeId ? messagesMap[activeId] ?? [] : [];
   230→  const activeTyping = activeId ? typingMap[activeId] === true : false;
   231→  const totalUnread = conversations.reduce((sum, c) => sum + c.unread, 0);
   232→
   233→  /* ---------------------------------------------------------------- */
   234→  /* Auto-scroll to latest message (scroll the viewport, never the page) */
   235→  /* ---------------------------------------------------------------- */
   236→  useEffect(() => {
   237→    const root = scrollRef.current;
   238→    if (!root) return;
   239→    const viewport = root.querySelector<HTMLDivElement>(
   240→      "[data-radix-scroll-area-viewport]"
   241→    );
   242→    if (viewport) viewport.scrollTop = viewport.scrollHeight;
   243→  }, [activeId, activeMessages, activeTyping]);
   244→
   245→  /* ---------------------------------------------------------------- */
   246→  /* Actions                                                           */
   247→  /* ---------------------------------------------------------------- */
   248→  const handleAuth = () => {
   249→    const socket = socketRef.current;
   250→    const trimmed = name.trim();
   251→    if (!socket || !connected || !trimmed) return;
   252→    setAuthError(null);
   253→    socket.emit("user:auth", { name: trimmed }, (res: UserAuthAck | ChatErrorAck) => {
   254→      if (res.ok) {
   255→        const next = { userId: res.user.id, name: res.user.name };
   256→        window.localStorage.setItem(MESSENGER_STORAGE_KEY, JSON.stringify(next));
   257→        meRef.current = next;
   258→        setMe(next);
   259→        setConversations(res.conversations);
   260→      } else {
   261→        setAuthError(
   262→          res.error === "INVALID_NAME"
   263→            ? "Nama tidak valid (1–40 karakter)."
   264→            : "Terjadi kesalahan, coba lagi."
   265→        );
   266→      }
   267→    });
   268→  };
   269→
   270→  const handleLogout = () => {
   271→    window.localStorage.removeItem(MESSENGER_STORAGE_KEY);
   272→    meRef.current = null;
   273→    activeIdRef.current = null;
   274→    setMe(null);
   275→    setConversations([]);
   276→    setMessagesMap({});
   277→    setTypingMap({});
   278→    setActiveId(null);
   279→    setFilter("");
   280→    setAuthError(null);
   281→    setInput("");
   282→    setSendError(false);
   283→    setNewChatOpen(false);
   284→    // Fresh socket ⇒ server cleanly forgets this client's rooms.
   285→    setEpoch((e) => e + 1);
   286→  };
   287→
   288→  const handleSelectConversation = (id: string) => {
   289→    const socket = socketRef.current;
   290→    if (!socket) return;
   291→    activeIdRef.current = id;
   292→    setActiveId(id);
   293→    setSendError(false);
   294→    socket.emit("messages:history", { conversationId: id }, (res: HistoryAck | ChatErrorAck) => {
   295→      if (res.ok) {
   296→        setMessagesMap((prev) => ({ ...prev, [id]: res.messages }));
   297→        socketRef.current?.emit("messages:read", { conversationId: id });
   298→      }
   299→    });
   300→  };
   301→
   302→  const handleBackToList = () => {
   303→    activeIdRef.current = null;
   304→    setActiveId(null);
   305→    setSendError(false);
   306→  };
   307→
   308→  const handleSend = () => {
   309→    const socket = socketRef.current;
   310→    const id = activeIdRef.current;
   311→    const content = input.trim();
   312→    if (!socket || !id || !connected || !content) return;
   313→    setInput("");
   314→    setSendError(false);
   315→    socket.emit(
   316→      "messages:send",
   317→      { conversationId: id, content },
   318→      (res: MessageAck | ChatErrorAck) => {
   319→        if (!res.ok) {
   320→          setInput(content); // restore the failed text
   321→          setSendError(true);
   322→        }
   323→      }
   324→    );
   325→  };
   326→
   327→  const handleInputChange = (value: string) => {
   328→    setInput(value);
   329→    setSendError(false);
   330→    const socket = socketRef.current;
   331→    const id = activeIdRef.current;
   332→    if (!socket || !id || !connected) return;
   333→    socket.emit("typing", { conversationId: id, isTyping: true });
   334→    if (typingTimerRef.current) clearTimeout(typingTimerRef.current);
   335→    typingTimerRef.current = setTimeout(() => {
   336→      socketRef.current?.emit("typing", { conversationId: id, isTyping: false });
   337→    }, 1500);
   338→  };
   339→
   340→  /* ---------------------------------------------------------------- */
   341→  /* Render: login                                                     */
   342→  /* ---------------------------------------------------------------- */
   343→  if (!me) {
   344→    return (
   345→      <div className="flex min-h-0 w-full flex-1 items-center justify-center px-4 pb-6">
   346→        <Card className="w-full max-w-md rounded-2xl">
   347→          <CardHeader>
   348→            <span
   349→              className="w-fit rounded-lg bg-emerald-600/10 p-3 text-emerald-600"
   350→              aria-hidden="true"
   351→            >
   352→              <MessageCircleMore className="size-6" />
   353→            </span>
   354→            <CardTitle className="text-xl">Masuk Chat</CardTitle>
   355→            <CardDescription>
   356→              Masukkan nama Anda untuk melihat percakapan dan chat dengan teman
   357→            </CardDescription>
   358→          </CardHeader>
   359→          <CardContent>
   360→            <form
   361→              className="space-y-3"
   362→              onSubmit={(e) => {
   363→                e.preventDefault();
   364→                handleAuth();
   365→              }}
   366→            >
   367→              <div className="space-y-2">
   368→                <Label htmlFor="messenger-name">Nama Anda</Label>
   369→                <Input
   370→                  id="messenger-name"
   371→                  value={name}
   372→                  maxLength={MAX_NAME_LENGTH}
   373→                  placeholder="cth. Budi Santoso"
   374→                  autoComplete="name"
   375→                  className="h-11"
   376→                  onChange={(e) => {
   377→                    setName(e.target.value);
   378→                    setAuthError(null);
   379→                  }}
   380→                />
   381→              </div>
   382→              {authError ? <p className="text-sm text-destructive">{authError}</p> : null}
   383→              <Button
   384→                type="submit"
   385→                className="h-11 w-full bg-emerald-600 text-white hover:bg-emerald-600/90"
   386→                disabled={!connected || !name.trim()}
   387→              >
   388→                {connected ? "Masuk" : "Menghubungkan…"}
   389→              </Button>
   390→              <p className="text-center text-xs text-muted-foreground">
   391→                Nama yang sama = akun yang sama, jadi Anda bisa lanjut chat kapan saja
   392→              </p>
   393→            </form>
   394→          </CardContent>
   395→        </Card>
   396→      </div>
   397→    );
   398→  }
   399→
   400→  /* ---------------------------------------------------------------- */
   401→  /* Render: messenger (full screen)                                   */
   402→  /* ---------------------------------------------------------------- */
   403→  const showSidebar = !isMobile || !activeId;
   404→  const showChatPane = !isMobile || activeId !== null;
   405→
   406→  const partnerStatus = activeTyping
   407→    ? "sedang mengetik…"
   408→    : activeConversation?.partner.online
   409→      ? "Online"
   410→      : "Offline";
   411→
   412→  return (
   413→    <div className="flex min-h-0 w-full flex-1 flex-col">
   414→      <div className="flex min-h-0 flex-1 flex-col overflow-hidden bg-card">
   415→        {/* Reconnecting strip */}
   416→        {!connected ? (
   417→          <p className="bg-amber-500/10 px-3 py-1.5 text-xs text-amber-600 dark:text-amber-400">
   418→            Koneksi terputus — mencoba menyambung ulang…
   419→          </p>
   420→        ) : null}
   421→
   422→        <div className="grid min-h-0 flex-1 grid-cols-[minmax(0,1fr)] md:grid-cols-[340px_1fr] lg:grid-cols-[380px_1fr]">
   423→          {/* ------------------------- Sidebar ------------------------- */}
   424→          {showSidebar ? (
   425→            <aside className="flex min-h-0 min-w-0 flex-col overflow-hidden md:border-r">
   426→              {/* Profile */}
   427→              <div className="flex items-center gap-3 border-b p-3">
   428→                <Avatar className="size-10">
   429→                  <AvatarFallback
   430→                    className={cn(
   431→                      "text-sm font-semibold text-white",
   432→                      avatarColorClass(me.name)
   433→                    )}
   434→                  >
   435→                    {initials(me.name)}
   436→                  </AvatarFallback>
   437→                </Avatar>
   438→                <div className="min-w-0 flex-1">
   439→                  <p className="truncate font-semibold leading-tight">{me.name}</p>
   440→                  <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   441→                    <span
   442→                      aria-hidden="true"
   443→                      className={cn(
   444→                        "inline-block size-1.5 shrink-0 rounded-full",
   445→                        connected ? "bg-emerald-500" : "bg-muted-foreground/40"
   446→                      )}
   447→                    />
   448→                    {connected ? "Online" : "Menghubungkan…"}
   449→                  </p>
   450→                </div>
   451→                <Button
   452→                  variant="ghost"
   453→                  size="icon"
   454→                  className="size-11 shrink-0 text-muted-foreground hover:text-destructive"
   455→                  aria-label="Keluar"
   456→                  onClick={handleLogout}
   457→                >
   458→                  <LogOut className="size-4" aria-hidden="true" />
   459→                </Button>
   460→              </div>
   461→
   462→              {/* Actions */}
   463→              <div className="flex items-center gap-2 p-3">
   464→                <Button
   465→                  className="h-11 flex-1 bg-emerald-600 text-white hover:bg-emerald-600/90"
   466→                  onClick={() => setNewChatOpen(true)}
   467→                >
   468→                  <UserRound className="size-4" aria-hidden="true" />
   469→                  Chat Baru
   470→                  {totalUnread > 0 ? (
   471→                    <Badge className="ml-1 bg-white px-1.5 text-[10px] text-emerald-700 hover:bg-white">
   472→                      {totalUnread}
   473→                    </Badge>
   474→                  ) : null}
   475→                </Button>
   476→              </div>
   477→
   478→              {/* Filter */}
   479→              <div className="px-3 pb-3">
   480→                <div className="relative">
   481→                  <Search
   482→                    className="pointer-events-none absolute left-3 top-1/2 size-4 -translate-y-1/2 text-muted-foreground"
   483→                    aria-hidden="true"
   484→                  />
   485→                  <Input
   486→                    value={filter}
   487→                    placeholder="Cari percakapan…"
   488→                    aria-label="Cari percakapan"
   489→                    className="h-10 pl-9"
   490→                    onChange={(e) => setFilter(e.target.value)}
   491→                  />
   492→                </div>
   493→              </div>
   494→
   495→              {/* Conversation list */}
   496→              <div className="chat-scroll min-h-0 flex-1">
   497→                <ScrollArea className="h-full">
   498→                  <div className="flex flex-col gap-1 p-2">
   499→                    {filteredConversations.length === 0 ? (
   500→                      <p className="px-3 py-8 text-center text-sm text-muted-foreground">
   501→                        {conversations.length === 0
   502→                          ? "Belum ada percakapan. Ketuk “Chat Baru” untuk mulai!"
   503→                          : "Tidak ada hasil."}
   504→                      </p>
   505→                    ) : (
   506→                      filteredConversations.map((c) => (
   507→                        <button
   508→                          key={c.id}
   509→                          type="button"
   510→                          onClick={() => handleSelectConversation(c.id)}
   511→                          aria-current={activeId === c.id ? "true" : undefined}
   512→                          className={cn(
   513→                            "flex w-full items-center gap-3 rounded-lg p-3 text-left transition-colors hover:bg-accent focus-visible:bg-accent",
   514→                            activeId === c.id && "bg-accent"
   515→                          )}
   516→                        >
   517→                          <span className="relative mt-0.5 shrink-0">
   518→                            <Avatar className="size-10">
   519→                              <AvatarFallback
   520→                                className={cn(
   521→                                  "text-sm font-semibold text-white",
   522→                                  avatarColorClass(c.partner.name)
   523→                                )}
   524→                              >
   525→                                {initials(c.partner.name)}
   526→                              </AvatarFallback>
   527→                            </Avatar>
   528→                            <span
   529→                              aria-label={c.partner.online ? "Online" : "Offline"}
   530→                              className={cn(
   531→                                "absolute -right-0.5 -bottom-0.5 size-3 rounded-full border-2 border-background",
   532→                                c.partner.online ? "bg-emerald-500" : "bg-muted-foreground/40"
   533→                              )}
   534→                            />
   535→                          </span>
   536→                          <span className="min-w-0 flex-1">
   537→                            <span className="flex items-center justify-between gap-2">
   538→                              <span className="truncate text-sm font-medium">
   539→                                {c.partner.name}
   540→                              </span>
   541→                              <span className="shrink-0 text-[10px] text-muted-foreground">
   542→                                {formatChatTime(c.lastMessageAt)}
   543→                              </span>
   544→                            </span>
   545→                            <span className="mt-0.5 flex items-center justify-between gap-2">
   546→                              <span className="truncate text-xs text-muted-foreground">
   547→                                {c.lastMessage
   548→                                  ? `${c.lastMessage.senderId === me.userId ? "Anda: " : ""}${c.lastMessage.content}`
   549→                                  : "Belum ada pesan"}
   550→                              </span>
   551→                              {c.unread > 0 ? (
   552→                                <Badge className="shrink-0 bg-emerald-600 px-1.5 text-[10px] text-white hover:bg-emerald-600">
   553→                                  {c.unread}
   554→                                </Badge>
   555→                              ) : null}
   556→                            </span>
   557→                          </span>
   558→                        </button>
   559→                      ))
   560→                    )}
   561→                  </div>
   562→                </ScrollArea>
   563→              </div>
   564→            </aside>

   567→          {/* ------------------------ Chat pane ------------------------ */}
