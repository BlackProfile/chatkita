     1→/**
     2→ * ChatKita chat-service — socket.io mini service (port 3003)
     3→ *
     4→ * Model (v7 — pure private messenger, like WhatsApp/Telegram):
     5→ *   users(role, pin_hash) ──< conversations ──< messages
     6→ *     messages: type text|image|voice|file|system, reply_to_id, duration_ms,
     7→ *               transcript (AI speech-to-text), deleted_at (soft delete),
     8→ *               edited_at/translation, file_name/file_size/mime_type
     9→ *               (file messages only)
    10→ *   + per-user read state (reads) → ✓✓ receipts broadcast live
    11→ *   + settings table (only Web Push VAPID keys are stored server-side)
    12→ *
    13→ * Every person who opens the app logs in with a name and gets a private
    14→ * 1-on-1 chat with the Admin account (the app owner), like messaging any
    15→ * contact. The Admin sees ALL conversations in one list, like WhatsApp Web.
    16→ *
    17→ * v7 — file sharing (documents, video, audio, archives, …):
    18→ *   Large files are uploaded OUT-OF-BAND. The bytes never touch this
    19→ *   service: the client POSTs them to the Next.js HTTP route POST /api/upload
    20→ *   (stored on disk under db/media/<random-name>), which returns the public
    21→ *   URL path "/api/media/<storedName>". The chat message then carries
    22→ *   type 'file' with that path as `content` plus the metadata
    23→ *   fileName/fileSize/mimeType, which this service validates (path-only
    24→ *   URL matching /api/media/<name>, name 1–255 chars, type/subtype mime,
    25→ *   size ≤ 25 MiB) and stores in messages.file_name/file_size/mime_type.
    26→ *   GET /api/media/<name> (Next.js route) streams the bytes back for
    27→ *   preview/download. Web Push bodies for file messages render as
    28→ *   "📎 <fileName>" via snippetOf.
    29→ *
    30→ * Feature set (pure messenger):
    31→ *   - Text | image | voice | file messages (+ replies, media size caps)
    32→ *   - Emoji reactions (message_reactions, one per user per message)
    33→ *   - Edit own text messages (15-min window, edited_at marker)
    34→ *   - Delete for everyone (soft delete + content redaction)
    35→ *   - Live ✓✓ read receipts (`read:update` broadcast)
    36→ *   - Presence (users' online state visible to the admins room ONLY)
    37→ *   - Typing relay between the two participants
    38→ *   - Pin a message per conversation (banner on both sides)
    39→ *   - Archive / unarchive conversations (a new user message auto-unarchives)
    40→ *   - Voice-note transcription (z-ai-web-dev-sdk ASR → message:updated)
    41→ *   - On-demand AI translation (messages.translation, cached)
    42→ *   - Web Push notifications when the recipient has no live socket
    43→ *     (VAPID keys generated once, stored in settings; subscriptions in
    44→ *     push_subscriptions; dead endpoints pruned on 404/410)
    45→ *   - Optional account PIN (SHA-256) protecting name-only logins
    46→ *
    47→ * Removed in v6 (v4/v5 customer-service tooling, no longer here): CRM
    48→ * labels/notes, pre-chat topics, operating hours, quick replies, SLA
    49→ * alerts, chatbot menu, star ratings, broadcast, CSV/print export,
    50→ * stats, AI auto-reply/suggestions/summary, `public:settings:update`.
    51→ * Historical SQLite columns (users.label/note/topic, messages.kind) and
    52→ * the old ratings table are left in existing databases but unused.
    53→ *
    54→ * Persistence: bun:sqlite (WAL mode) at ./chat.db
    55→ * Connection path MUST stay '/' — the Caddy gateway forwards
    56→ * /?XTransformPort=3003 to this port.
    57→ */

   369→/** Directory size helper (bytes) for the dashboard storage panel. */
   370→const dirStats = (dir: string): { bytes: number; files: number } => {
   371→  let bytes = 0
   372→  let files = 0
   373→  try {
   374→    for (const entry of readdirSync(dir)) {
   375→      try {
   376→        const st = statSync(join(dir, entry))
   377→        if (st.isFile()) {
   378→          bytes += st.size
   379→          files += 1
   380→        }
   381→      } catch {
   382→        /* file vanished mid-scan */
   383→      }
   384→    }
   385→  } catch {
   386→    /* directory missing yet */
   387→  }
   388→  return { bytes, files }
   389→}

   391→/**
   392→ * v10 — aggregated dashboard stats for `admin:dashboard`.
   393→ * All aggregation happens in SQL; days are UTC epoch-day buckets.
   394→ */
   395→const dashboardStats = () => {
   396→  const nowMs = now()
   397→  const DAY_MS = 86_400_000
   398→
   399→  const scalar = (sql: string, ...params: (string | number)[]): number =>
   400→    Number((db.query(sql).get(...params) as { v: number | null } | undefined)?.v ?? 0)
   401→
   402→  const users = scalar("SELECT COUNT(*) AS v FROM users WHERE role = 'user'")
   403→  const conversations = scalar('SELECT COUNT(*) AS v FROM conversations')
   404→  const messages = scalar('SELECT COUNT(*) AS v FROM messages')
   405→  const deletedMessages = scalar('SELECT COUNT(*) AS v FROM messages WHERE deleted_at IS NOT NULL')
   406→  const last24h = scalar('SELECT COUNT(*) AS v FROM messages WHERE created_at >= ?', nowMs - DAY_MS)
   407→  const last7d = scalar('SELECT COUNT(*) AS v FROM messages WHERE created_at >= ?', nowMs - 7 * DAY_MS)
   408→
   409→  const byTypeRows = db
   410→    .query('SELECT type, COUNT(*) AS c FROM messages GROUP BY type')
   411→    .all() as { type: string | null; c: number }[]
   412→  const byType: Record<string, number> = { text: 0, image: 0, voice: 0, file: 0, system: 0 }
   413→  for (const r of byTypeRows) {
   414→    const key = r.type && r.type in byType ? r.type : 'text'
   415→    byType[key] += r.c
   416→  }
   417→
   418→  // Daily series — last 14 UTC days, zero-filled.
   419→  const dailyRows = db
   420→    .query(
   421→      `SELECT CAST(created_at / 86400000 AS INTEGER) AS day, COUNT(*) AS c
   422→       FROM messages WHERE created_at >= ? GROUP BY day`
   423→    )
   424→    .all(nowMs - 13 * DAY_MS) as { day: number; c: number }[]
   425→  const dailyMap = new Map(dailyRows.map((r) => [r.day, r.c]))
   426→  const daily: { date: string; count: number }[] = []
   427→  for (let i = 13; i >= 0; i -= 1) {
   428→    const epochDay = Math.floor(nowMs / DAY_MS) - i
   429→    daily.push({
   430→      date: new Date(epochDay * DAY_MS).toISOString().slice(0, 10),
   431→      count: dailyMap.get(epochDay) ?? 0,
   432→    })
   433→  }
   434→
   435→  // Hour-of-day distribution over the last 7 days (0-23, UTC).
   436→  const hourlyRows = db
   437→    .query(
   438→      `SELECT CAST(strftime('%H', created_at / 1000, 'unixepoch') AS INTEGER) AS h, COUNT(*) AS c
   439→       FROM messages WHERE created_at >= ? GROUP BY h`
   440→    )
   441→    .all(nowMs - 7 * DAY_MS) as { h: number; c: number }[]
   442→  const hourly: number[] = Array.from({ length: 24 }, () => 0)
   443→  for (const r of hourlyRows) {
   444→    if (r.h >= 0 && r.h < 24) hourly[r.h] = r.c
   445→  }
   446→
   447→  const topUsers = (
   448→    db
   449→      .query(
   450→        `SELECT u.id, u.name, u.last_seen_at,
   451→           (SELECT COUNT(*) FROM messages m WHERE m.sender_id = u.id) AS c
   452→         FROM users u WHERE u.role = 'user'
   453→         ORDER BY c DESC, u.name ASC LIMIT 10`
   454→      )
   455→      .all() as { id: string; name: string; last_seen_at: number; c: number }[]
   456→  ).map((r) => ({
   457→    id: r.id,
   458→    name: r.name,
   459→    messages: r.c,
   460→    lastSeenAt: new Date(r.last_seen_at).toISOString(),
   461→    online: isOnline(r.id),
   462→  }))
   463→
   464→  const allUsers = (
   465→    db
   466→      .query(
   467→        `SELECT u.id, u.name, u.created_at, u.last_seen_at,
   468→           (SELECT COUNT(*) FROM messages m WHERE m.sender_id = u.id) AS c
   469→         FROM users u WHERE u.role = 'user'
   470→         ORDER BY u.last_seen_at DESC LIMIT 100`
   471→      )
   472→      .all() as { id: string; name: string; created_at: number; last_seen_at: number; c: number }[]
   473→  ).map((r) => ({
   474→    id: r.id,
   475→    name: r.name,
   476→    joinedAt: new Date(r.created_at).toISOString(),
   477→    lastSeenAt: new Date(r.last_seen_at).toISOString(),
   478→    messages: r.c,
   479→    online: isOnline(r.id),
   480→  }))
   481→
   482→  const mediaRow = db
   483→    .query(
   484→      `SELECT COUNT(*) AS c, COALESCE(SUM(file_size), 0) AS b FROM messages
   485→       WHERE file_size IS NOT NULL AND deleted_at IS NULL AND media_expired_at IS NULL`
   486→    )
   487→    .get() as { c: number; b: number }
   488→
   489→  let dbBytes = 0
   490→  let walBytes = 0
   491→  try {
   492→    dbBytes = statSync(DB_PATH).size
   493→    walBytes = statSync(`${DB_PATH}-wal`).size
   494→  } catch {
   495→    /* WAL may not exist */
   496→  }
   497→  const media = dirStats(MEDIA_DIR)
   498→
   499→  let onlineUsers = 0
   500→  for (const [id, sockets] of onlineSockets) {
   501→    if (id !== ADMIN_ID && sockets.size > 0) onlineUsers += 1
   502→  }
   503→
   504→  return {
   505→    generatedAt: new Date(nowMs).toISOString(),
   506→    version: SERVICE_VERSION,
   507→    uptimeMs: nowMs - BOOT_AT,
   508→    totals: {
   509→      users,
   510→      conversations,
   511→      messages,
   512→      deletedMessages,
   513→      last24h,
   514→      last7d,
   515→      byType,
   516→      onlineUsers,
   517→      mediaCount: Number(mediaRow.c),
   518→      mediaBytes: Number(mediaRow.b),
   519→    },
   520→    daily,
   521→    hourly,
   522→    topUsers,
   523→    users: allUsers,
   524→    storage: {
   525→      dbBytes,
   526→      walBytes,
   527→      mediaBytes: media.bytes,
   528→      mediaFiles: media.files,
   529→      quotaBytes: QUOTA_BYTES,
   530→      retentionDays: Math.round(RETENTION_MS / DAY_MS),
   531→    },
   532→  }
   533→}

   563→/** Fire-and-forget push to every stored subscription of `userId`. */

   565→  if (!VAPID_PUBLIC) return

   579→        } catch (err) {
   580→          const status = (err as { statusCode?: number }).statusCode
   581→          if (status === 404 || status === 410) {
   582→            db.run('DELETE FROM push_subscriptions WHERE endpoint = ?', [s.endpoint])

   585→      })

   587→  } catch (err) {

  1330→      args.length > 0 && typeof args[0] === 'object' && args[0] !== null ? args[0] : {}
