   100→/** Read (and validate) the persisted login. Client-side only. */
   101→function readStoredUser(): StoredUser | null {
   102→  try {
   103→    const raw = window.localStorage.getItem(CHAT_SESSION_KEY);
   104→    if (!raw) return null;
   105→    const parsed = JSON.parse(raw) as Partial<StoredUser> | null;
   106→    if (
   107→      parsed &&
   108→      typeof parsed.userId === "string" &&
   109→      parsed.userId.length > 0 &&
   110→      typeof parsed.name === "string" &&
   111→      parsed.name.length > 0
   112→    ) {
   113→      return { userId: parsed.userId, name: parsed.name };
   114→    }
   115→    return null;
   116→  } catch {
   117→    return null;
   118→  }
   119→}

   165→/** Downscale + compress an image file to a JPEG data URL the server accepts. */
   166→async function fileToDataUrl(file: File): Promise<string> {
   167→  if (file.size > 6 * 1024 * 1024) throw new Error("too-large");
   168→  const bitmap = await createImageBitmap(file);
   169→  const max = 1280;
   170→  const scale = Math.min(1, max / Math.max(bitmap.width, bitmap.height));
   171→  const canvas = document.createElement("canvas");
   172→  canvas.width = Math.round(bitmap.width * scale);
   173→  canvas.height = Math.round(bitmap.height * scale);
   174→  const ctx = canvas.getContext("2d");
   175→  if (!ctx) throw new Error("canvas");
   176→  ctx.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
   177→  bitmap.close();
   178→  return canvas.toDataURL("image/jpeg", 0.82);
   179→}

   260→
   261→function PinDialog({
   262→  open,
   263→  onOpenChange,
   264→  socketRef,
   265→  hasPin,
   266→  onHasPin,
   267→}: {
   268→  open: boolean;
   269→  onOpenChange: (v: boolean) => void;
   270→  socketRef: React.RefObject<Socket | null>;
   271→  hasPin: boolean;
   272→  onHasPin: (v: boolean) => void;
   273→}) {
   274→  const [pin, setPin] = useState("");
   275→  const [error, setError] = useState<string | null>(null);
   276→  const [saved, setSaved] = useState(false);
   277→
   278→  // Mounted only while open (see render below) — state resets on remount.
   279→
   280→  const submit = () => {
   281→    const socket = socketRef.current;
   282→    if (!socket || !/^\d{4,8}$/.test(pin)) {
   283→      setError("PIN harus 4–8 angka.");
   284→      return;
   285→    }
   286→    socket.emit(
   287→      "user:setpin",
   288→      { pin },
   289→      (res: AckOf<SetPinAck>) => {
   290→        if (res.ok) {
   291→          onHasPin(res.hasPin);
   292→          setSaved(true);
   293→          setError(null);
   294→          setTimeout(() => onOpenChange(false), 700);
   295→        } else {
   296→          setError(res.error === "INVALID_PIN" ? "PIN harus 4–8 angka." : "Gagal, coba lagi.");
   297→        }
   298→      }
   299→    );
   300→  };
   301→
   302→  const clear = () => {
   303→    const socket = socketRef.current;
   304→    if (!socket) return;
   305→    socket.emit("user:setpin", { pin: null }, (res: AckOf<SetPinAck>) => {
   306→      if (res.ok) {
   307→        onHasPin(false);
   308→        onOpenChange(false);
   309→      } else setError("Gagal menghapus PIN.");
   310→    });
   311→  };
   312→
   313→  return (
   314→    <Dialog open={open} onOpenChange={onOpenChange}>
   315→      <DialogContent className="max-w-sm rounded-2xl">
   316→        <DialogHeader>
   317→          <DialogTitle className="flex items-center gap-2">
   318→            <ShieldCheck className="size-5 text-emerald-600" aria-hidden="true" />
   319→            Kunci Akun dengan PIN
   320→          </DialogTitle>
   321→          <DialogDescription>
   322→            {hasPin
   323→              ? "Akun Anda terlindungi PIN. Ganti atau hapus PIN di sini."
   324→              : "Tanpa PIN, siapa pun yang tahu nama Anda bisa membuka chat ini. Pasang PIN 4–8 angka agar lebih aman."}
   325→          </DialogDescription>
   326→        </DialogHeader>
   327→        {saved ? (
   328→          <p className="text-sm font-medium text-emerald-600">PIN tersimpan ✓</p>
   329→        ) : (
   330→          <div className="space-y-3">
   331→            <div className="space-y-2">
   332→              <Label htmlFor="pin-entry">PIN baru (4–8 angka)</Label>
   333→              <Input
   334→                id="pin-entry"
   335→                type="password"
   336→                inputMode="numeric"
   337→                maxLength={8}
   338→                value={pin}
   339→                placeholder="••••"
   340→                onChange={(e) => {
   341→                  setPin(e.target.value.replace(/\D/g, ""));
   342→                  setError(null);
   343→                }}
   344→                onKeyDown={(e) => {
   345→                  if (e.key === "Enter") submit();
   346→                }}
   347→              />
   348→              {error ? <p className="text-sm text-destructive">{error}</p> : null}
   349→            </div>
   350→            <div className="flex gap-2">
   351→              <Button
   352→                className="h-10 flex-1 bg-emerald-600 text-white hover:bg-emerald-600/90"
   353→                onClick={submit}
   354→              >
   355→                Simpan PIN
   356→              </Button>
   357→              {hasPin ? (
   358→                <Button variant="outline" className="h-10" onClick={clear}>
   359→                  Hapus PIN
   360→                </Button>
   361→              ) : null}
   362→            </div>
   363→          </div>
   364→        )}
   365→      </DialogContent>
   366→    </Dialog>
   367→  );
   368→}
