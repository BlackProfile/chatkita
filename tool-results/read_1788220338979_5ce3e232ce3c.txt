     1→     1→/**
     2→     2→ * ChatKita chat-service — socket.io mini service (port 3003)
     3→     3→ *
     4→     4→ * Model (v7 — pure private messenger, like WhatsApp/Telegram):
     5→     5→ *   users(role, pin_hash) ──< conversations ──< messages
     6→     6→ *     messages: type text|image|voice|file|system, reply_to_id, duration_ms,
     7→     7→ *               transcript (AI speech-to-text), deleted_at (soft delete),
     8→     8→ *               edited_at/translation, file_name/file_size/mime_type
     9→     9→ *               (file messages only)
    10→    10→ *   + per-user read state (reads) → ✓✓ receipts broadcast live
    11→    11→ *   + settings table (only Web Push VAPID keys are stored server-side)
    12→    12→ *
    13→    13→ * Every person who opens the app logs in with a name and gets a private
    14→    14→ * 1-on-1 chat with the Admin account (the app owner), like messaging any
    15→    15→ * contact. The Admin sees ALL conversations in one list, like WhatsApp Web.
    16→    16→ *
    17→    17→ * v7 — file sharing (documents, video, audio, archives, …):
    18→    18→ *   Large files are uploaded OUT-OF-BAND. The bytes never touch this
    19→    19→ *   service: the client POSTs them to the Next.js HTTP route POST /api/upload
    20→    20→ *   (stored on disk under db/media/<random-name>), which returns the public
    21→    21→ *   URL path "/api/media/<storedName>". The chat message then carries
    22→    22→ *   type 'file' with that path as `content` plus the metadata
    23→    23→ *   fileName/fileSize/mimeType, which this service validates (path-only
    24→    24→ *   URL matching /api/media/<name>, name 1–255 chars, type/subtype mime,
    25→    25→ *   size ≤ 25 MiB) and stores in messages.file_name/file_size/mime_type.
    26→    26→ *   GET /api/media/<name> (Next.js route) streams the bytes back for
    27→    27→ *   preview/download. Web Push bodies for file messages render as
    28→    28→ *   "📎 <fileName>" via snippetOf.
    29→    29→ *
    30→    30→ * Feature set (pure messenger):
    31→    31→ *   - Text | image | voice | file messages (+ replies, media size caps)
    32→    32→ *   - Emoji reactions (message_reactions, one per user per message)
    33→    33→ *   - Edit own text messages (15-min window, edited_at marker)
    34→    34→ *   - Delete for everyone (soft delete + content redaction)
    35→    35→ *   - Live ✓✓ read receipts (`read:update` broadcast)
    36→    36→ *   - Presence (users' online state visible to the admins room ONLY)
    37→    37→ *   - Typing relay between the two participants
    38→    38→ *   - Pin a message per conversation (banner on both sides)
    39→    39→ *   - Archive / unarchive conversations (a new user message auto-unarchives)
    40→    40→ *   - Voice-note transcription (z-ai-web-dev-sdk ASR → message:updated)
    41→    41→ *   - On-demand AI translation (messages.translation, cached)
    42→    42→ *   - Web Push notifications when the recipient has no live socket
    43→    43→ *     (VAPID keys generated once, stored in settings; subscriptions in
    44→    44→ *     push_subscriptions; dead endpoints pruned on 404/410)
    45→    45→ *   - Optional account PIN (SHA-256) protecting name-only logins
    46→    46→ *
    47→    47→ * Removed in v6 (v4/v5 customer-service tooling, no longer here): CRM
    48→    48→ * labels/notes, pre-chat topics, operating hours, quick replies, SLA

   391→   391→/**
   392→   392→ * v10 — aggregated dashboard stats for `admin:dashboard`.
   393→   393→ * All aggregation happens in SQL; days are UTC epoch-day buckets.
   394→   394→ */
   395→   395→const dashboardStats = () => {
   396→   396→  const nowMs = now()
   397→   397→  const DAY_MS = 86_400_000
   398→   398→
   399→   399→  const scalar = (sql: string, ...params: (string | number)[]): number =>
   400→   400→    Number((db.query(sql).get(...params) as { v: number | null } | undefined)?.v ?? 0)
   401→   401→
   402→   402→  const users = scalar("SELECT COUNT(*) AS v FROM users WHERE role = 'user'")
   403→   403→  const conversations = scalar('SELECT COUNT(*) AS v FROM conversations')
   404→   404→  const messages = scalar('SELECT COUNT(*) AS v FROM messages')
   405→   405→  const deletedMessages = scalar('SELECT COUNT(*) AS v FROM messages WHERE deleted_at IS NOT NULL')
   406→   406→  const last24h = scalar('SELECT COUNT(*) AS v FROM messages WHERE created_at >= ?', nowMs - DAY_MS)
   407→   407→  const last7d = scalar('SELECT COUNT(*) AS v FROM messages WHERE created_at >= ?', nowMs - 7 * DAY_MS)
   408→   408→
   409→   409→  const byTypeRows = db
   410→   410→    .query('SELECT type, COUNT(*) AS c FROM messages GROUP BY type')
   411→   411→    .all() as { type: string | null; c: number }[]
   412→   412→  const byType: Record<string, number> = { text: 0, image: 0, voice: 0, file: 0, system: 0 }
   413→   413→  for (const r of byTypeRows) {
   414→   414→    const key = r.type && r.type in byType ? r.type : 'text'
   415→   415→    byType[key] += r.c
   416→   416→  }
   417→   417→
   418→   418→  // Daily series — last 14 UTC days, zero-filled.
   419→   419→  const dailyRows = db
   420→   420→    .query(
   421→   421→      `SELECT CAST(created_at / 86400000 AS INTEGER) AS day, COUNT(*) AS c
   422→   422→       FROM messages WHERE created_at >= ? GROUP BY day`
   423→   423→    )
   424→   424→    .all(nowMs - 13 * DAY_MS) as { day: number; c: number }[]
   425→   425→  const dailyMap = new Map(dailyRows.map((r) => [r.day, r.c]))
   426→   426→  const daily: { date: string; count: number }[] = []
   427→   427→  for (let i = 13; i >= 0; i -= 1) {
   428→   428→    const epochDay = Math.floor(nowMs / DAY_MS) - i
   429→   429→    daily.push({
   430→   430→      date: new Date(epochDay * DAY_MS).toISOString().slice(0, 10),
   431→   431→      count: dailyMap.get(epochDay) ?? 0,
   432→   432→    })
   433→   433→  }
   434→   434→
   435→   435→  // Hour-of-day distribution over the last 7 days (0-23, UTC).
   436→   436→  const hourlyRows = db
   437→   437→    .query(
   438→   438→      `SELECT CAST(strftime('%H', created_at / 1000, 'unixepoch') AS INTEGER) AS h, COUNT(*) AS c
   439→   439→       FROM messages WHERE created_at >= ? GROUP BY h`
   440→   440→    )
   441→   441→    .all(nowMs - 7 * DAY_MS) as { h: number; c: number }[]
   442→   442→  const hourly: number[] = Array.from({ length: 24 }, () => 0)
   443→   443→  for (const r of hourlyRows) {
   444→   444→    if (r.h >= 0 && r.h < 24) hourly[r.h] = r.c
   445→   445→  }
   446→   446→
   447→   447→  const topUsers = (
   448→   448→    db
