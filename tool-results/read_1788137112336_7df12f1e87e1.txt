   377→   377→/** Fire-and-forget push to every stored subscription of `userId`. */
   378→   378→const pushSend = async (userId: string, payload: PushPayload) => {
   379→   379→  if (!VAPID_PUBLIC) return
   380→   380→  try {
   381→   381→    const subs = db
   382→   382→      .query('SELECT * FROM push_subscriptions WHERE user_id = ?')
   383→   383→      .all(userId) as Array<{ endpoint: string; p256dh: string; auth: string }>
   384→   384→    if (subs.length === 0) return
   385→   385→    await Promise.all(
   386→   386→      subs.map(async (s) => {
   387→   387→        try {
   388→   388→          await webpush.sendNotification(
   389→   389→            { endpoint: s.endpoint, keys: { p256dh: s.p256dh, auth: s.auth } },
   390→   390→            JSON.stringify(payload),
   391→   391→            { TTL: 3600 }
   392→   392→          )
   393→   393→        } catch (err) {
   394→   394→          const status = (err as { statusCode?: number }).statusCode
   395→   395→          if (status === 404 || status === 410) {
   396→   396→            db.run('DELETE FROM push_subscriptions WHERE endpoint = ?', [s.endpoint])
   397→   397→          }
   398→   398→        }
   399→   399→      })
   400→   400→    )
   401→   401→  } catch (err) {
   402→   402→    console.error('Push send error:', (err as Error)?.message ?? err)
   403→   403→  }
   404→   404→}
