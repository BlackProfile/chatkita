    91→/** Read (and validate) the persisted login. Client-side only. */
    92→function readStoredUser(): StoredUser | null {
    93→  try {
    94→    const raw = window.localStorage.getItem(CHAT_SESSION_KEY);
    95→    if (!raw) return null;
    96→    const parsed = JSON.parse(raw) as Partial<StoredUser> | null;
    97→    if (
    98→      parsed &&
    99→      typeof parsed.userId === "string" &&
   100→      parsed.userId.length > 0 &&
   101→      typeof parsed.name === "string" &&
   102→      parsed.name.length > 0
   103→    ) {
   104→      return { userId: parsed.userId, name: parsed.name };
   105→    }
   106→    return null;
   107→  } catch {
   108→    return null;
   109→  }
   110→}

   156→/** Downscale + compress an image file to a JPEG data URL the server accepts. */
   157→async function fileToDataUrl(file: File): Promise<string> {
   158→  if (file.size > 6 * 1024 * 1024) throw new Error("too-large");
   159→  const bitmap = await createImageBitmap(file);
   160→  const max = 1280;
   161→  const scale = Math.min(1, max / Math.max(bitmap.width, bitmap.height));
   162→  const canvas = document.createElement("canvas");
   163→  canvas.width = Math.round(bitmap.width * scale);
   164→  canvas.height = Math.round(bitmap.height * scale);
   165→  const ctx = canvas.getContext("2d");
   166→  if (!ctx) throw new Error("canvas");
   167→  ctx.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
   168→  bitmap.close();
   169→  return canvas.toDataURL("image/jpeg", 0.82);
   170→}

   181→function PinDialog({
   182→  open,
   183→  onOpenChange,
   184→  socketRef,
   185→  hasPin,
   186→  onHasPin,
   187→}: {
   188→  open: boolean;
   189→  onOpenChange: (v: boolean) => void;
   190→  socketRef: React.RefObject<Socket | null>;
   191→  hasPin: boolean;
   192→  onHasPin: (v: boolean) => void;
   193→}) {
   194→  const [pin, setPin] = useState("");
   195→  const [error, setError] = useState<string | null>(null);
   196→  const [saved, setSaved] = useState(false);
   197→
   198→  // Mounted only while open (see render below) — state resets on remount.
   199→
   200→  const submit = () => {
   201→    const socket = socketRef.current;
   202→    if (!socket || !/^\d{4,8}$/.test(pin)) {
   203→      setError("PIN harus 4–8 angka.");
   204→      return;
   205→    }
   206→    socket.emit(
   207→      "user:setpin",
   208→      { pin },
   209→      (res: AckOf<SetPinAck>) => {
   210→        if (res.ok) {
   211→          onHasPin(res.hasPin);
   212→          setSaved(true);
   213→          setError(null);
   214→          setTimeout(() => onOpenChange(false), 700);
   215→        } else {
   216→          setError(res.error === "INVALID_PIN" ? "PIN harus 4–8 angka." : "Gagal, coba lagi.");
   217→        }
   218→      }
   219→    );
   220→  };
   221→
   222→  const clear = () => {
   223→    const socket = socketRef.current;
   224→    if (!socket) return;
   225→    socket.emit("user:setpin", { pin: null }, (res: AckOf<SetPinAck>) => {
   226→      if (res.ok) {
   227→        onHasPin(false);
   228→        onOpenChange(false);
   229→      } else setError("Gagal menghapus PIN.");
   230→    });
   231→  };
   232→
   233→  return (
   234→    <Dialog open={open} onOpenChange={onOpenChange}>
   235→      <DialogContent className="max-w-sm rounded-2xl">
   236→        <DialogHeader>
   237→          <DialogTitle className="flex items-center gap-2">
   238→            <ShieldCheck className="size-5 text-emerald-600" aria-hidden="true" />
   239→            Kunci Akun dengan PIN
   240→          </DialogTitle>
   241→          <DialogDescription>
   242→            {hasPin
   243→              ? "Akun Anda terlindungi PIN. Ganti atau hapus PIN di sini."
   244→              : "Tanpa PIN, siapa pun yang tahu nama Anda bisa membuka chat ini. Pasang PIN 4–8 angka agar lebih aman."}
   245→          </DialogDescription>
   246→        </DialogHeader>
   247→        {saved ? (
   248→          <p className="text-sm font-medium text-emerald-600">PIN tersimpan ✓</p>
   249→        ) : (
   250→          <div className="space-y-3">
   251→            <div className="space-y-2">
   252→              <Label htmlFor="pin-entry">PIN baru (4–8 angka)</Label>
   253→              <Input
   254→                id="pin-entry"
   255→                type="password"
   256→                inputMode="numeric"
   257→                maxLength={8}
   258→                value={pin}
   259→                placeholder="••••"
   260→                onChange={(e) => {
   261→                  setPin(e.target.value.replace(/\D/g, ""));
   262→                  setError(null);
   263→                }}
   264→                onKeyDown={(e) => {
   265→                  if (e.key === "Enter") submit();
   266→                }}
   267→              />
   268→              {error ? <p className="text-sm text-destructive">{error}</p> : null}
   269→            </div>
   270→            <div className="flex gap-2">
   271→              <Button
   272→                className="h-10 flex-1 bg-emerald-600 text-white hover:bg-emerald-600/90"
   273→                onClick={submit}
   274→              >
   275→                Simpan PIN
   276→              </Button>
   277→              {hasPin ? (
   278→                <Button variant="outline" className="h-10" onClick={clear}>
   279→                  Hapus PIN
   280→                </Button>
   281→              ) : null}
   282→            </div>
   283→          </div>
   284→        )}
   285→      </DialogContent>
   286→    </Dialog>
   287→  );
   288→}

   310→  // so reading localStorage in the initializer is safe.

   324→  const [emojiOpen, setEmojiOpen] = useState(false);
   325→  const [pendingImage, setPendingImage] = useState<string | null>(null);
   326→  const [imageError, setImageError] = useState<string | null>(null);
   327→  const [searchOpen, setSearchOpen] = useState(false);
   328→  const [searchQuery, setSearchQuery] = useState("");
   329→  const [showJump, setShowJump] = useState(false);
   330→  const [newCount, setNewCount] = useState(0);
   331→  const [lightbox, setLightbox] = useState<string | null>(null);
   332→
   333→  // Web Push VAPID public key (null = push unavailable).
   334→  const [pushPublicKey, setPushPublicKey] = useState<string | null>(null);
   335→  const [pinnedMsg, setPinnedMsg] = useState<{ id: number; snippet: string } | null>(null);
   336→  const [editing, setEditing] = useState<ChatMessage | null>(null);
   337→  const [fontScale, setFontScale] = useState<FontScale>(() => readFontScale());
   338→  const [installAvailable, setInstallAvailable] = useState(false);
   339→  const [translatingId, setTranslatingId] = useState<number | null>(null);
   340→
   341→  /** Bumped on logout to tear down + recreate the socket (fresh rooms). */
   342→  const [epoch, setEpoch] = useState(0);
   343→
   344→  const socketRef = useRef<Socket | null>(null);
   345→  const meRef = useRef<StoredUser | null>(null);
   346→  const conversationIdRef = useRef<string | null>(null);
   347→  const typingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   348→  const partnerTypingTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   349→  const scrollRef = useRef<HTMLDivElement | null>(null);

   361→
   362→  /* ---------------------------------------------------------------- */
   363→  /* Socket lifecycle (recreated on logout via `epoch`)                */
   364→  /* ---------------------------------------------------------------- */
   365→  useEffect(() => {
   366→    // Restore persisted login (ref only) so the first `connect` re-auths
   367→    // immediately. State is set inside the connect ack.
   368→    meRef.current = readStoredUser();
   369→
   370→    const socket = createChatSocket();
   371→    socketRef.current = socket;
   372→
   373→    const loadHistory = (id: string) => {
   374→      socket.emit(
   375→        "messages:history",
   376→        { conversationId: id },
   377→        (res: AckOf<HistoryAck>) => {
   378→          if (res.ok) {
   379→            setPartner(res.partner);
   380→            setMessages(res.messages);
   381→            setAdminReadId(res.partnerLastReadId);
   382→            setPinnedMsg(res.pinned ? { id: res.pinned.id, snippet: res.pinned.snippet } : null);
   383→            socketRef.current?.emit("messages:read", { conversationId: id });
   384→          }
   385→        }
   386→      );
   387→    };
   388→
   389→    socket.on("connect", () => {
   390→      setConnected(true);
   391→      // Pre-login: fetch the Web Push VAPID public key (v6 shape:
   392→      // { ok, pushPublicKey }) so it is ready right after login.

   394→        socket.emit("public:settings", {}, (res: AckOf<PublicSettingsAck>) => {

   399→      if (!current) return;
   400→      // Re-auth on EVERY connect to (re)join the personal room and get
   401→      // the freshest history (also closes reconnect gaps).

   404→        { name: current.name, userId: current.userId },

   410→            setLastName(res.user.name);

   414→            setHasPin(res.user.hasPin ?? false);
   415→            setNeedsPin(false);
   416→            setPinEntry("");

   418→            setPartner(res.partner);
   419→            setMessages(res.messages);
   420→            setAdminReadId(res.partnerLastReadId);

   431→            window.localStorage.removeItem(CHAT_SESSION_KEY);

   433→            conversationIdRef.current = null;

   435→            setConversationId(null);
   436→            setPartner(null);
   437→            setMessages([]);

   440→                ? "Nama “Admin” tidak tersedia — coba nama lain."
   441→                : "Sesi berakhir. Silakan masuk kembali."
   442→            );
   443→          }
   444→        }

   471→    // has the same id (history-replacement vs live-event race).

   479→        conversationId: msg.conversationId,

   487→        hiddenUnreadRef.current += 1;

   489→        setTitleUnread(hiddenUnreadRef.current);
   490→      }
   491→    });
   492→
   493→    // Delete tombstones + late voice transcripts + edits/translations/reactions.
   494→    socket.on("message:updated", (u: MessageUpdatePayload) => {
   495→      setMessages((prev) =>
   496→        prev.map((m) =>
   497→          m.id === u.id
   498→            ? {
   499→                ...m,
   500→                content: u.content ?? m.content,
   501→                deletedAt: u.deletedAt ?? m.deletedAt,
   502→                transcript: u.transcript ?? m.transcript,
   503→                editedAt: u.editedAt ?? m.editedAt,
   504→                translation: u.translation ?? m.translation,
   505→                reactions: u.reactions ?? m.reactions,
   506→              }
   507→            : m
   508→        )
   509→      );
   510→      if (u.translation && u.id === translatingIdRef.current) {
   511→        translatingIdRef.current = null;
   512→        setTranslatingId(null);
   513→      }
   514→    });
   515→
   516→    // Live ✓✓: the admin read up to `lastReadMessageId`.
   517→    socket.on(
   518→      "read:update",
   519→      (r: { conversationId: string; userId: string; lastReadMessageId: number }) => {
   520→        if (r.userId === ADMIN_ID) setAdminReadId(r.lastReadMessageId);
   521→      }
   522→    );
   523→
   524→    socket.on(
   525→      "partner:typing",
   526→      ({ isTyping }: { conversationId: string; isTyping: boolean }) => {
   527→        if (partnerTypingTimerRef.current) {
   528→          clearTimeout(partnerTypingTimerRef.current);
   529→          partnerTypingTimerRef.current = null;
   530→        }
   531→        setPartnerTyping(isTyping);
   532→        if (isTyping) {
   533→          // Auto-clear after 8s in case a stop event is missed.
   534→          partnerTypingTimerRef.current = setTimeout(() => {
   535→            partnerTypingTimerRef.current = null;
   536→            setPartnerTyping(false);
   537→          }, 8000);
   538→        }
   539→      }
   540→    );
   541→
   542→    // Users only ever receive presence for the admin (server-enforced).
   543→    socket.on(
   544→      "presence:update",
   545→      ({
   546→        userId,
   547→        online,
   548→        lastSeenAt,
   549→      }: {
   550→        userId: string;
   551→        online: boolean;
   552→        lastSeenAt: string | null;
   553→      }) => {
   554→        if (userId !== ADMIN_ID) return;
   555→        setPartner((prev) =>
   556→          prev
   557→            ? { ...prev, online, lastSeenAt: online ? null : lastSeenAt }
   558→            : prev
   559→        );
   560→      }
   561→    );
   562→
   563→    // Leaving the tab clears the unread title badge.
   564→    const onVisible = () => {
   565→      if (!document.hidden) {
   566→        hiddenUnreadRef.current = 0;
   567→        setTitleUnread(0);
   568→      }
   569→    };
   570→    document.addEventListener("visibilitychange", onVisible);
