   100→   100→/** Read (and validate) the persisted login. Client-side only. */
   101→   101→function readStoredUser(): StoredUser | null {
   102→   102→  try {
   103→   103→    const raw = window.localStorage.getItem(CHAT_SESSION_KEY);
   104→   104→    if (!raw) return null;
   105→   105→    const parsed = JSON.parse(raw) as Partial<StoredUser> | null;
   106→   106→    if (
   107→   107→      parsed &&
   108→   108→      typeof parsed.userId === "string" &&
   109→   109→      parsed.userId.length > 0 &&
   110→   110→      typeof parsed.name === "string" &&
   111→   111→      parsed.name.length > 0
   112→   112→    ) {
   113→   113→      return { userId: parsed.userId, name: parsed.name };
   114→   114→    }
   115→   115→    return null;
   116→   116→  } catch {
   117→   117→    return null;
   118→   118→  }
   119→   119→}

   165→   165→/** Downscale + compress an image file to a JPEG data URL the server accepts. */
   166→   166→async function fileToDataUrl(file: File): Promise<string> {
   167→   167→  if (file.size > 6 * 1024 * 1024) throw new Error("too-large");
   168→   168→  const bitmap = await createImageBitmap(file);
   169→   169→  const max = 1280;
   170→   170→  const scale = Math.min(1, max / Math.max(bitmap.width, bitmap.height));
   171→   171→  const canvas = document.createElement("canvas");
   172→   172→  canvas.width = Math.round(bitmap.width * scale);
   173→   173→  canvas.height = Math.round(bitmap.height * scale);
   174→   174→  const ctx = canvas.getContext("2d");
   175→   175→  if (!ctx) throw new Error("canvas");
   176→   176→  ctx.drawImage(bitmap, 0, 0, canvas.width, canvas.height);
   177→   177→  bitmap.close();
   178→   178→  return canvas.toDataURL("image/jpeg", 0.82);
   179→   179→}

   261→   261→function PinDialog({
   262→   262→  open,
   263→   263→  onOpenChange,
   264→   264→  socketRef,
   265→   265→  hasPin,
   266→   266→  onHasPin,
   267→   267→}: {
   268→   268→  open: boolean;
   269→   269→  onOpenChange: (v: boolean) => void;
   270→   270→  socketRef: React.RefObject<Socket | null>;
   271→   271→  hasPin: boolean;
   272→   272→  onHasPin: (v: boolean) => void;
   273→   273→}) {
   274→   274→  const [pin, setPin] = useState("");
   275→   275→  const [error, setError] = useState<string | null>(null);
   276→   276→  const [saved, setSaved] = useState(false);
   277→   277→
   278→   278→  // Mounted only while open (see render below) — state resets on remount.
   279→   279→
   280→   280→  const submit = () => {
   281→   281→    const socket = socketRef.current;
   282→   282→    if (!socket || !/^\d{4,8}$/.test(pin)) {
   283→   283→      setError("PIN harus 4–8 angka.");
   284→   284→      return;
   285→   285→    }
   286→   286→    socket.emit(
   287→   287→      "user:setpin",
   288→   288→      { pin },
   289→   289→      (res: AckOf<SetPinAck>) => {
   290→   290→        if (res.ok) {
   291→   291→          onHasPin(res.hasPin);
   292→   292→          setSaved(true);
   293→   293→          setError(null);
   294→   294→          setTimeout(() => onOpenChange(false), 700);
   295→   295→        } else {
   296→   296→          setError(res.error === "INVALID_PIN" ? "PIN harus 4–8 angka." : "Gagal, coba lagi.");
   297→   297→        }
   298→   298→      }
   299→   299→    );
   300→   300→  };
   301→   301→
   302→   302→  const clear = () => {
   303→   303→    const socket = socketRef.current;
   304→   304→    if (!socket) return;
   305→   305→    socket.emit("user:setpin", { pin: null }, (res: AckOf<SetPinAck>) => {
   306→   306→      if (res.ok) {
   307→   307→        onHasPin(false);
   308→   308→        onOpenChange(false);
   309→   309→      } else setError("Gagal menghapus PIN.");
   310→   310→    });
   311→   311→  };
   312→   312→
   313→   313→  return (
   314→   314→    <Dialog open={open} onOpenChange={onOpenChange}>
   315→   315→      <DialogContent className="max-w-sm rounded-2xl">
   316→   316→        <DialogHeader>
   317→   317→          <DialogTitle className="flex items-center gap-2">
   318→   318→            <ShieldCheck className="size-5 text-emerald-600" aria-hidden="true" />
   319→   319→            Kunci Akun dengan PIN
   320→   320→          </DialogTitle>
   321→   321→          <DialogDescription>
   322→   322→            {hasPin
   323→   323→              ? "Akun Anda terlindungi PIN. Ganti atau hapus PIN di sini."
   324→   324→              : "Tanpa PIN, siapa pun yang tahu nama Anda bisa membuka chat ini. Pasang PIN 4–8 angka agar lebih aman."}
   325→   325→          </DialogDescription>
   326→   326→        </DialogHeader>
   327→   327→        {saved ? (
   328→   328→          <p className="text-sm font-medium text-emerald-600">PIN tersimpan ✓</p>
   329→   329→        ) : (
   330→   330→          <div className="space-y-3">
   331→   331→            <div className="space-y-2">
   332→   332→              <Label htmlFor="pin-entry">PIN baru (4–8 angka)</Label>
   333→   333→              <Input
   334→   334→                id="pin-entry"
   335→   335→                type="password"
   336→   336→                inputMode="numeric"
   337→   337→                maxLength={8}
   338→   338→                value={pin}
   339→   339→                placeholder="••••"
   340→   340→                onChange={(e) => {
   341→   341→                  setPin(e.target.value.replace(/\D/g, ""));
   342→   342→                  setError(null);
   343→   343→                }}
   344→   344→                onKeyDown={(e) => {
   345→   345→                  if (e.key === "Enter") submit();
   346→   346→                }}
   347→   347→              />
   348→   348→              {error ? <p className="text-sm text-destructive">{error}</p> : null}
   349→   349→            </div>
   350→   350→            <div className="flex gap-2">
   351→   351→              <Button
   352→   352→                className="h-10 flex-1 bg-emerald-600 text-white hover:bg-emerald-600/90"
   353→   353→                onClick={submit}
   354→   354→              >
   355→   355→                Simpan PIN
   356→   356→              </Button>
   357→   357→              {hasPin ? (
   358→   358→                <Button variant="outline" className="h-10" onClick={clear}>
   359→   359→                  Hapus PIN
   360→   360→                </Button>
   361→   361→              ) : null}
   362→   362→            </div>
   363→   363→          </div>
   364→   364→        )}
   365→   365→      </DialogContent>
   366→   366→    </Dialog>
   367→   367→  );
   368→   368→}
