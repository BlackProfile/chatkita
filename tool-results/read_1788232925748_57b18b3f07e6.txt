    99→    99→const fmtUptime = (ms: number): string => {
   100→   100→  const total = Math.max(0, Math.floor(ms / 1000));
   101→   101→  const d = Math.floor(total / 86400);
   102→   102→  const h = Math.floor((total % 86400) / 3600);
   103→   103→  const m = Math.floor((total % 3600) / 60);
   104→   104→  if (d > 0) return `${d} hari ${h} jam`;
   105→   105→  if (h > 0) return `${h} jam ${m} menit`;
   106→   106→  return `${m} menit`;
   107→   107→};
   108→   108→
   109→   109→const fmtDay = (iso: string): string => {
   110→   110→  const [, m, d] = iso.split("-");
   111→   111→  return `${d}/${m}`;
   112→   112→};
   113→   113→
   114→   114→const fmtDateTime = (iso: string): string => {
   115→   115→  const d = new Date(iso);
   116→   116→  if (Number.isNaN(d.getTime())) return "—";
   117→   117→  return d.toLocaleString("id-ID", { dateStyle: "medium", timeStyle: "short" });
   118→   118→};

   181→   181→export function AdminDashboard({
   182→   182→  open,
   183→   183→  onOpenChange,
   184→   184→  socket,
   185→   185→  tab,
   186→   186→  onTabChange,
   187→   187→}: {
   188→   188→  open: boolean;
   189→   189→  onOpenChange: (open: boolean) => void;
   190→   190→  /** Socket admin yang sudah terautentikasi. */
   191→   191→  socket: Socket | null;
   192→   192→  tab: DashboardTab;
   193→   193→  onTabChange: (tab: DashboardTab) => void;
   194→   194→}) {
   195→   195→  const [stats, setStats] = useState<DashboardStats | null>(null);
   196→   196→  const [loading, setLoading] = useState(false);
   197→   197→  const [settings, setSettings] = useState<AppSettings | null>(null);
   198→   198→  const [settingsMsg, setSettingsMsg] = useState<string | null>(null);
   199→   199→  const [broadcastText, setBroadcastText] = useState("");
   200→   200→  const [broadcastKind, setBroadcastKind] = useState<"siaran" | "pengumuman">("siaran");
   201→   201→  const [broadcastMsg, setBroadcastMsg] = useState<string | null>(null);
   202→   202→  const [broadcasting, setBroadcasting] = useState(false);
   203→   203→  const [vacuumResult, setVacuumResult] = useState<string | null>(null);
   204→   204→  const [busy, setBusy] = useState<"backup" | "vacuum" | null>(null);
   205→   205→
   206→   206→  const fetchStats = useCallback(() => {
   207→   207→    if (!socket || !socket.connected) return;
   208→   208→    setLoading(true);
   209→   209→    socket.emit("admin:dashboard", {}, (res: AckOf<DashboardStatsAck>) => {
   210→   210→      setLoading(false);
   211→   211→      if (res.ok) setStats(res.stats);
   212→   212→    });
   213→   213→  }, [socket]);
   214→   214→
   215→   215→  // Muat stats + settings saat dialog dibuka (deferred — bukan setState
   216→   216→  // sinkron di body effect) + auto-refresh ringan tiap 30 detik.
   217→   217→  useEffect(() => {
   218→   218→    if (!open) return;
   219→   219→    const t = setTimeout(() => {
   220→   220→      fetchStats();
   221→   221→      if (socket?.connected) {
   222→   222→        socket.emit("admin:settings:get", {}, (res: AckOf<AppSettingsAck>) => {
   223→   223→          if (res.ok) setSettings(res.settings);
   224→   224→        });
   225→   225→      }
   226→   226→    }, 0);
   227→   227→    const iv = setInterval(fetchStats, 30_000);
   228→   228→    return () => {
   229→   229→      clearTimeout(t);
   230→   230→      clearInterval(iv);
   231→   231→    };
   232→   232→  }, [open, fetchStats, socket]);
   233→   233→
   234→   234→  const totals = stats?.totals;
   235→   235→
   236→   236→  const typeEntries = useMemo(() => {
   237→   237→    if (!totals) return [];
   238→   238→    return Object.entries(totals.byType).filter(([, v]) => v > 0);
   239→   239→  }, [totals]);
   240→   240→  const typeMax = Math.max(1, ...typeEntries.map(([, v]) => v));
   241→   241→  const typeTotal = typeEntries.reduce((s, [, v]) => s + v, 0);
   242→   242→
   243→   243→  const saveSettings = (patch: Partial<AppSettings>) => {
   244→   244→    if (!socket) return;
   245→   245→    setSettingsMsg(null);
   246→   246→    socket.emit(
   247→   247→      "admin:settings:set",
   248→   248→      patch,
   249→   249→      (res: AckOf<AppSettingsAck> | ChatErrorAck) => {
   250→   250→        if (res.ok) {
   251→   251→          setSettings(res.settings);
   252→   252→          setSettingsMsg("Tersimpan ✓");
   253→   253→          setTimeout(() => setSettingsMsg(null), 2000);
   254→   254→        } else {
   255→   255→          setSettingsMsg("Gagal menyimpan");
   256→   256→        }
   257→   257→      }
   258→   258→    );
   259→   259→  };
   260→   260→
   261→   261→  const sendBroadcast = () => {
   262→   262→    const text = broadcastText.trim();
   263→   263→    if (!socket || !text || broadcasting) return;
   264→   264→    setBroadcasting(true);
   265→   265→    setBroadcastMsg(null);
   266→   266→    socket.emit(
   267→   267→      "admin:broadcast",
   268→   268→      { text, kind: broadcastKind },
   269→   269→      (res: AckOf<BroadcastAck>) => {
   270→   270→        setBroadcasting(false);
   271→   271→        if (res.ok) {
   272→   272→          setBroadcastMsg(`Terkirim ke ${res.count} percakapan ✓`);
   273→   273→          setBroadcastText("");
   274→   274→          setTimeout(() => setBroadcastMsg(null), 3000);
   275→   275→        } else {
   276→   276→          setBroadcastMsg("Gagal mengirim");
   277→   277→        }
   278→   278→      }
   279→   279→    );
   280→   280→  };
   281→   281→
   282→   282→  const downloadBackup = () => {
   283→   283→    if (!socket || busy) return;
   284→   284→    setBusy("backup");
   285→   285→    socket.emit("admin:backup", {}, (res: AckOf<BackupAck>) => {
   286→   286→      setBusy(null);
   287→   287→      if (!res.ok) return;
   288→   288→      try {
   289→   289→        const blob = new Blob([JSON.stringify(res, null, 2)], {
   290→   290→          type: "application/json",
   291→   291→        });
   292→   292→        const url = URL.createObjectURL(blob);
   293→   293→        const a = document.createElement("a");
   294→   294→        a.href = url;
   295→   295→        a.download = `chatkita-backup-${res.exportedAt.slice(0, 10)}.json`;
   296→   296→        a.click();
   297→   297→        URL.revokeObjectURL(url);
   298→   298→      } catch {
   299→   299→        /* ignore */
   300→   300→      }
   301→   301→    });
   302→   302→  };
   303→   303→
   304→   304→  const runVacuum = () => {
   305→   305→    if (!socket || busy) return;
   306→   306→    setBusy("vacuum");
   307→   307→    setVacuumResult(null);
   308→   308→    socket.emit("admin:vacuum", {}, (res: AckOf<VacuumAck>) => {
   309→   309→      setBusy(null);
   310→   310→      if (!res.ok) return;
   311→   311→      const saved = res.before.walBytes - res.after.walBytes;
   312→   312→      setVacuumResult(
   313→   313→        `Selesai — DB ${(res.after.dbBytes / 1024).toFixed(0)} KB, WAL ${(res.after.walBytes / 1024).toFixed(0)} KB${saved > 0 ? ` (hemat ${(saved / 1024).toFixed(0)} KB)` : ""}`
   314→   314→      );
   315→   315→      fetchStats();
   316→   316→      setTimeout(() => setVacuumResult(null), 5000);
   317→   317→    });
   318→   318→  };
   319→   319→
   320→   320→  /** Baris pengguna (tabel tab Pengguna / papan top users). */
   321→   321→  const userRow = (u: DashboardUserRow, rank?: number) => (
   322→   322→    <div
   323→   323→      key={u.id}
   324→   324→      className="flex items-center gap-3 rounded-xl px-2 py-2 transition-colors hover:bg-accent/50"
   325→   325→    >
   326→   326→      <span className="relative shrink-0">
   327→   327→        <Avatar className="size-9">
   328→   328→          <AvatarFallback
   329→   329→            className={cn("text-xs font-semibold text-white", avatarColorClass(u.name))}
   330→   330→          >
   331→   331→            {initials(u.name)}
   332→   332→          </AvatarFallback>
   333→   333→        </Avatar>
   334→   334→        <span
   335→   335→          aria-hidden="true"
   336→   336→          className={cn(
   337→   337→            "absolute -right-0.5 -bottom-0.5 size-2.5 rounded-full border-2 border-background",
   338→   338→            u.online ? "bg-emerald-500" : "bg-muted-foreground/40"
   339→   339→          )}
   340→   340→        />
   341→   341→      </span>
   342→   342→      <div className="min-w-0 flex-1">
   343→   343→        <p className="flex items-center gap-1.5 truncate text-sm font-medium">
   344→   344→          {rank === 1 ? "🥇 " : rank === 2 ? "🥈 " : rank === 3 ? "🥉 " : ""}
   345→   345→          {u.name}
   346→   346→        </p>
   347→   347→        <p className="truncate text-[11px] text-muted-foreground">
   348→   348→          {u.online ? "Online" : `Terakhir dilihat ${formatLastSeen(u.lastSeenAt)}`}
   349→   349→        </p>
   350→   350→      </div>
   351→   351→      <div className="shrink-0 text-right">
   352→   352→        <p className="text-sm font-semibold tabular-nums">{u.messages}</p>
   353→   353→        <p className="text-[10px] text-muted-foreground">pesan</p>
   354→   354→      </div>
   355→   355→    </div>
   356→   356→  );
   357→   357→
   358→   358→  return (
   359→   359→    <Dialog open={open} onOpenChange={onOpenChange}>
   360→   360→      <DialogContent className="flex max-h-[92vh] w-[calc(100vw-1rem)] max-w-3xl flex-col gap-0 overflow-hidden rounded-2xl p-0 sm:max-w-5xl">
   361→   361→        <DialogHeader className="shrink-0 border-b bg-muted/30 px-4 py-3 sm:px-6 sm:py-4">
   362→   362→          <DialogTitle className="flex items-center gap-2 text-base sm:text-lg">
   363→   363→            <span className="flex size-8 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500 to-teal-600 text-white shadow-sm shadow-emerald-600/25">
   364→   364→              <GaugeCircle className="size-4.5" aria-hidden="true" />
   365→   365→            </span>
   366→   366→            Dashboard Aplikasi
   367→   367→            {stats ? (
   368→   368→              <Badge variant="secondary" className="ml-1 font-mono text-[10px]">
   369→   369→                {stats.version}
   370→   370→              </Badge>
   371→   371→            ) : null}
   372→   372→          </DialogTitle>
   373→   373→          <DialogDescription className="sr-only">
   374→   374→            Analitik, pengaturan, dan pengelolaan aplikasi ChatKita.
   375→   375→          </DialogDescription>
   376→   376→        </DialogHeader>
   377→   377→
   378→   378→        {/* Tab bar */}
   379→   379→        <div className="flex shrink-0 gap-1 overflow-x-auto border-b px-3 py-2 sm:px-6">
   380→   380→          {TABS.map((t) => (
   381→   381→            <button
   382→   382→              key={t.key}
   383→   383→              type="button"
   384→   384→              aria-pressed={tab === t.key}
   385→   385→              onClick={() => onTabChange(t.key)}
   386→   386→              className={cn(
   387→   387→                "flex h-8 shrink-0 items-center gap-1.5 rounded-full px-3 text-xs font-medium transition-all",
   388→   388→                tab === t.key
   389→   389→                  ? "bg-gradient-to-br from-emerald-500 to-emerald-600 text-white shadow-sm shadow-emerald-600/25"
   390→   390→                  : "bg-muted/60 text-muted-foreground hover:bg-accent hover:text-foreground"
   391→   391→              )}
   392→   392→            >
   393→   393→              <t.icon className="size-3.5" aria-hidden="true" />
   394→   394→              {t.label}
   395→   395→            </button>
   396→   396→          ))}
   397→   397→        </div>
   398→   398→
   399→   399→        {/* Konten tab */}
   400→   400→        <div className="chat-scroll min-h-0 flex-1 overflow-y-auto overscroll-contain p-3 sm:p-5">
   401→   401→          {!stats && loading ? (
   402→   402→            <div className="flex h-40 items-center justify-center text-muted-foreground">
   403→   403→              <Loader2 className="size-5 animate-spin" aria-hidden="true" />
   404→   404→            </div>
   405→   405→          ) : !stats ? (
   406→   406→            <p className="py-10 text-center text-sm text-muted-foreground">
   407→   407→              Data belum tersedia.
   408→   408→            </p>
   409→   409→          ) : tab === "ringkasan" ? (
   410→   410→            <div className="space-y-3">
   411→   411→              <div className="grid grid-cols-2 gap-2.5 lg:grid-cols-3">
   412→   412→                <Kpi
   413→   413→                  icon={Users}
   414→   414→                  label="Total pengguna"
   415→   415→                  value={String(totals!.users)}
   416→   416→                  sub={`${totals!.onlineUsers} online sekarang`}
   417→   417→                />
   418→   418→                <Kpi
   419→   419→                  icon={MessagesSquare}
   420→   420→                  label="Percakapan"
   421→   421→                  value={String(totals!.conversations)}
   422→   422→                />
   423→   423→                <Kpi
   424→   424→                  icon={MessageSquare}
   425→   425→                  label="Pesan 24 jam"
   426→   426→                  value={String(totals!.last24h)}
   427→   427→                  sub={`${totals!.last7d} dalam 7 hari`}

   434→   434→                />
   435→   435→                <Kpi
   436→   436→                  icon={HardDrive}
   437→   437→                  label="Media tersimpan"
   438→   438→                  value={formatFileSize(totals!.mediaBytes)}
   439→   439→                  sub={`${totals!.mediaCount} file aktif`}
   440→   440→                />
   441→   441→                <Kpi
   442→   442→                  icon={Clock}
   443→   443→                  label="Uptime service"
   444→   444→                  value={fmtUptime(stats.uptimeMs)}
   445→   445→                />
   446→   446→              </div>
   447→   447→
   448→   448→              <BarChart
   449→   449→                values={stats.daily.map((d) => d.count)}
   450→   450→                labels={stats.daily.map((d) => fmtDay(d.date))}
   451→   451→                title="Pesan per hari (14 hari terakhir)"

   453→   453→
   454→   454→              <div className="rounded-xl border bg-card p-3">
